;; compiler/link
;; Wasm modules from C: a dynamic-linking side module (clang -shared, Emscripten's side modules) rewritten into the form
;; of a program's modules, on the prelude's memory, table, stack and allocator.
;; requires compiler/macros compiler/leb128 compiler/binary compiler/running

;; ---- the module's indices ----

;; link-side-module gives (bytes hooks): the module's bytes, and the names of the functions it imports by name. prelude
;; names the prelude in the program.
(define (link-side-module [what datum] [bytes datum] [prelude datum]) -> datum)

;; ==== private ====

(define (fail [what string] [args datum]))

(define (at [n u32]) -> datum)

(define (int [d datum]) -> u32)

(define (sleb) -> i64)

(define (lookup [l datum] [key datum]) -> datum)

(define (put! [q (ptr buffer)] [s (slice u8)]))

(define (put-name! [q (ptr buffer)] [s datum]))

(define (put-section! [out (ptr buffer)] [id u32] [q (ptr buffer)]))

(define (op! [t (ptr buffer)] [b u32]))

(define (op-u! [t (ptr buffer)] [b u32] [n u32]))

(define (i32-const! [t (ptr buffer)] [n i64]))

(define (fc! [t (ptr buffer)] [sub u32] [n u32]))

;; A side module as read. A type is its parameters' count; an import (module field kind from to type), its description
;; at from to to and type a function's; a defined global (from init); an element segment (mode offset items), mode
;; active, at the table base plus offset, passive or declarative; a data segment (placed offset from length); a body
;; (from . to); a custom section (name from to).
(struct side [bytes (slice u8)] [types datum] [type-bytes datum] [imports datum] [ftypes datum] [globals datum]
  [exports datum] [elems datum] [datas datum] [bodies datum] [customs datum] [start i64] [nfimports u32]
  [ngimports u32] [mem-size u32] [mem-align u32] [table-size u32])

(define (read-side! [s (ptr side)] [what datum]))

(define (custom! [s (ptr side)] [start u32] [size u32]))

(define (types-in! [s (ptr side)]))

(define (imports-in! [s (ptr side)] [what datum]))

(define (indexes-in) -> datum)

(define (globals-in! [s (ptr side)]))

(define (skip-init))

(define (exports-in! [s (ptr side)]))

(define (offset-expr [what datum]) -> i64)

(define (elems-in! [s (ptr side)] [what datum]))

(define (bodies-in! [s (ptr side)]))

(define (datas-in! [s (ptr side)] [what datum]))

;; ---- the module's indices ----

(define (linker-name? [f datum]) -> bool)

(define (prelude-name [f datum]) -> datum)

;; A link holds the new indices: of each old function and global in fmap and gmap, of the prelude's sbrk, of the
;; bases, of the start; the kept function imports and the hooks, each (import . old index); and the GOT entries, each
;; (kind value new), value the data's offset from the memory base or the old function index.
(struct link [fmap (ptr u32)] [gmap (ptr u32)] [nf u32] [ng u32] [kept datum] [hooks datum] [got datum] [sp datum]
  [sbrk u32] [mb u32] [tb u32] [first-slot u32] [start u32] [ndefined u32])

(define (global-value [s (ptr side)] [what datum] [g u32]) -> i64)

(define (exported-function [s (ptr side)] [f datum]) -> i64)

(define (got-entry [s (ptr side)] [what datum] [m datum] [f datum]) -> datum)

(define (indices! [k (ptr link)] [s (ptr side)] [what datum]))

(define (fmapped [k (ptr link)] [i u32]) -> u32)

(define (gmapped [k (ptr link)] [i u32]) -> u32)

;; ---- the module ----

(define (module-bytes [s (ptr side)] [k (ptr link)] [what datum] [prelude datum]) -> datum)

(define (imports! [q (ptr buffer)] [s (ptr side)] [k (ptr link)] [prelude datum] [sbrk-type u32]))

(define (globals! [q (ptr buffer)] [s (ptr side)] [k (ptr link)]))

(define (init! [q (ptr buffer)] [s (ptr side)] [k (ptr link)] [from u32]))

(define (exports! [q (ptr buffer)] [s (ptr side)] [k (ptr link)]))

(define (elements! [q (ptr buffer)] [s (ptr side)] [k (ptr link)]))

(define (datas! [q (ptr buffer)] [s (ptr side)]))

(define (code! [q (ptr buffer)] [s (ptr side)] [k (ptr link)] [what datum]))

(define (supplied! [q (ptr buffer)] [s (ptr side)] [k (ptr link)]))

(define (body! [q (ptr buffer)] [s (ptr side)] [k (ptr link)] [from u32] [to u32] [what datum]))

(define (stub! [q (ptr buffer)] [params u32] [slot u32] [type u32]))

(define (start! [q (ptr buffer)] [s (ptr side)] [k (ptr link)]))

;; ---- names ----

;; Name subsections 1, 2 and 3 are by function, 7 by global.
(define (names! [q (ptr buffer)] [s (ptr side)] [k (ptr link)] [from u32] [to u32]))
