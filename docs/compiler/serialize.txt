;; compiler/serialize
;; The serializer: data as bytes, in the format syntax.sub's deserialize reads.
;; requires compiler/macros compiler/leb128

(struct serializer [out buffer] [count u32] [shared (ptr hash-table)] [fixnums (ptr hash-table)]
        [atoms (ptr hash-table)] [pairs (ptr hash-table)] [scopes (ptr hash-table)] [table datum])

(define (make-serializer) -> (ptr serializer))

(define (serializer-scope-table [s (ptr serializer)]) -> datum)

(define (serialize! [s (ptr serializer)] [d datum]) -> u32)

(define (serializer-bytes [s (ptr serializer)]) -> datum)

;; ==== private ====

(define (scope-index! [s (ptr serializer)] [sc u32]) -> u32)

(define (added! [s (ptr serializer)] [t (ptr hash-table)] [key u64]) -> u32)

(define (shared-index [s (ptr serializer)] [key datum]) -> u32)

(define (shared-added! [s (ptr serializer)] [key datum]) -> u32)

(define (bytes! [p (ptr buffer)] [b (slice u8)]))

(define (small [n i64]) -> datum)

(define (serialize-list! [s (ptr serializer)] [d datum]) -> u32)

(define (serialize-syntax! [s (ptr serializer)] [d datum]) -> u32)

(define (scope-indexes [s (ptr serializer)] [set (ptr scope-set)]) -> datum)

(define (serialize-srcloc! [s (ptr serializer)] [l (ptr srcloc)]) -> u32)
