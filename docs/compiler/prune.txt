;; compiler/prune
;; The pruner removes the functions a program can never call, imported ones among them, and renumbers the rest.
;; requires compiler/macros compiler/leb128 compiler/binary compiler/running

;; ---- what the program can call ----

;; A module without Substrate's record, a C module, passes through as it is, and the exports it imports stay.
(define (pruned [all datum]) -> datum)

;; ---- writing ----

(define (stripped [modules datum]) -> datum)

;; ==== private ====

(struct wasm [name datum] [bytes (slice u8)] [sections datum] [imports datum] [spans datum] [funcs datum]
        [nimports u32] [exports datum] [by-name (ptr hash-table)] [start i32] [count u32] [starts (ptr u32)]
        [ends (ptr u32)] [sites buffer] [site-starts (ptr u32)] [site-ends (ptr u32)] [live (ptr bool)]
        [kept (ptr hash-table)] [indexes (ptr u32)] [global-imports datum] [global-exports (ptr hash-table)]
        [tables datum] [nglobals u32] [ndatas u32] [table-functions (ptr bool)] [read-globals (ptr bool)] [read-datas (ptr bool)]
        [valued (ptr bool)])

;; ---- reading ----

(define (unreadable))

(define (new-wasm [name datum] [bytes datum]) -> (ptr wasm))

(define (read-rest! [w (ptr wasm)]))

;; Each import's span is its kind and its bytes' start and end.
(define (read-imports! [w (ptr wasm)]))

(define (read-exports! [w (ptr wasm)]))

(define (read-elems!))

(define (read-bodies! [w (ptr wasm)]))

;; ---- what a body calls ----

;; calls-in returns the functions body k calls, and adds each function index's offset, length and value to sites, and
;; the globals and data segments it uses to read-globals and read-datas.
(define (calls-in [w (ptr wasm)] [k u32]) -> datum)

;; ---- what the program can call ----

(define work datum)

(define (live? [w (ptr wasm)] [i u32]) -> bool)

(define (mark! [w (ptr wasm)] [i u32]))

(define (wasm-named [ws datum] [name datum]) -> (ptr wasm))

(define (exported [w (ptr wasm)] [name datum]) -> i64)

(define (loads-code? [f datum]) -> bool)

(define (reach! [ws datum] [entry (ptr wasm)]) -> bool)

(define (table-function? [w (ptr wasm)] [i u32]) -> bool)

;; tables! keeps the function that copies in a global array of number literals (compile.sub's table!) when a kept
;; function reads the global.
(define (tables! [ws datum] [entry (ptr wasm)]))

(define (global-read? [ws datum] [entry (ptr wasm)] [w (ptr wasm)] [name datum]) -> bool)

(define (keep! [w (ptr wasm)] [name datum]))

(define (used? [w (ptr wasm)] [name datum]) -> bool)

;; These host functions never return bytes from the program's malloc.
(define giving-nothing datum)

;; keep-hooks! keeps the functions the host calls in the first module: on-event when the program imports from a host
;; module other than host, on-trap when it calls trap-handler!, and malloc when it imports a host function outside
;; giving-nothing or has an on-trap.
(define (keep-hooks! [ws datum] [entry (ptr wasm)]))

(define (merged [all datum] [opaque datum] [out datum]) -> datum)

(define (pruned-of [modules datum] [kept datum]) -> datum)

;; ---- the new indices ----

(define (renumber! [w (ptr wasm)]))

(define (renumbered [w (ptr wasm)] [i u32]) -> u32)

(define (write-uleb-padded [q (ptr buffer)] [n u32] [length u32]))

;; ---- writing ----

(define (write-name [q (ptr buffer)] [s datum]))

(define (section-out [out (ptr buffer)] [id i32] [q (ptr buffer)]))

(define (custom-name [w (ptr wasm)] [start u32]) -> datum)

(define (rewritten [w (ptr wasm)]) -> datum)

(define (without [w (ptr wasm)] [names datum]) -> datum)

(define (imports! [out (ptr buffer)] [w (ptr wasm)]))

(define (functions! [out (ptr buffer)] [w (ptr wasm)] [start u32] [end u32]))

;; A function whose value the program takes keeps its export, so that the host can find it, as a worker's is found.
(define (exports! [out (ptr buffer)] [w (ptr wasm)]))

(define (start! [out (ptr buffer)] [w (ptr wasm)]))

(define (elements! [out (ptr buffer)] [w (ptr wasm)] [start u32] [end u32]))

;; body! copies body k, rewriting each function index in as many bytes as before, and a call of a table function
;; tables! did not keep as nops.
(define (body! [q (ptr buffer)] [w (ptr wasm)] [k u32]))

;; datas! empties each data segment no kept function reads.
(define (datas! [out (ptr buffer)] [w (ptr wasm)] [start u32] [end u32]))

(define (code! [out (ptr buffer)] [w (ptr wasm)] [news (ptr u32)]))

(define (names! [out (ptr buffer)] [w (ptr wasm)] [start u32] [end u32]))

(define (locations! [out (ptr buffer)] [w (ptr wasm)] [start u32] [end u32] [news (ptr u32)]))
