;; ---- the allocator, memory functions, arenas, buffers, hash tables ----

;; A block is a word of its size, a multiple of 16, with 1 set while it is used, 2 while the block before it is, and 4
;; while it is in a quick list, then what malloc gives, at a multiple of 16. A free block also holds the next and
;; previous blocks in its list, and its size again in its last word; no two free blocks are neighbours. Each region from
;; sbrk ends in a used block of size 0, the last region's at heap-end.
(struct heap-block [head u32] [next (ptr heap-block)] [prev (ptr heap-block)])

;; The free lists: one for each size under 1 KB, then four from each power of two to the next. free-map has the bit of
;; each list that holds a block.
(define free-lists (array (ptr heap-block) 152))

(define free-map (array u32 5))

(define heap-end (ptr heap-block))

(define heap-limit u32)

;; The quick lists: a freed block of up to 4 KB waits in the list of its size, still used, for malloc to give it again;
;; before the heap grows, they are all freed.
(define quick-lists (array (ptr heap-block) 257))
