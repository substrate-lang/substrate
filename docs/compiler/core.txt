;; compiler/core
;; The records the expander and the back end share.
;; requires compiler/macros

(define (func-info [name datum] [params datum] [results datum] [instance datum] [export datum]) -> datum)

(define (func-info? [d datum]) -> bool)

(define (func-info-name [d datum]) -> datum)

(define (func-info-params [d datum]) -> datum)

(define (func-info-results [d datum]) -> datum)

(define (func-info-instance [d datum]) -> datum)

(define (func-info-export [d datum]) -> datum)

(define (set-func-info-instance! [d datum] [v datum]))

(define (set-func-info-export! [d datum] [v datum]))

(define (func-info-place [d datum]) -> i32)

(define (set-func-info-place! [d datum] [i i32]))

(define (global-info [name datum] [type datum] [mutable datum] [instance datum] [export datum]) -> datum)

(define (global-info? [d datum]) -> bool)

(define (global-info-name [d datum]) -> datum)

(define (global-info-type [d datum]) -> datum)

(define (global-info-mutable? [d datum]) -> datum)

(define (global-info-instance [d datum]) -> datum)

(define (global-info-export [d datum]) -> datum)

(define (set-global-info-instance! [d datum] [v datum]))

(define (set-global-info-export! [d datum] [v datum]))

(define (tag-info [name datum] [params datum] [instance datum] [export datum]) -> datum)

(define (tag-info? [d datum]) -> bool)

(define (tag-info-name [d datum]) -> datum)

(define (tag-info-params [d datum]) -> datum)

(define (tag-info-instance [d datum]) -> datum)

(define (tag-info-export [d datum]) -> datum)

(define (set-tag-info-instance! [d datum] [v datum]))

(define (set-tag-info-export! [d datum] [v datum]))

(define (keys-from! [n i64]) -> i64)

(define (fresh-key!) -> datum)

;; ==== private ====

(const func-kind u32 1)

(const global-kind u32 2)

(const tag-kind u32 3)

;; A function's place is its index among its module's functions, or -1 if the module keeps no slots.
(struct func-record [self datum] [name datum] [params datum] [results datum] [instance datum] [export datum] [place i32])

(struct global-record [self datum] [name datum] [type datum] [mutable datum] [instance datum] [export datum])

(struct tag-record [self datum] [name datum] [params datum] [instance datum] [export datum])

(define (new-record [kind u32] [size u32]) -> (ptr void))

(define (checked [d datum] [kind u32] [what string]) -> (ptr void))

(define (func-of [d datum]) -> (ptr func-record))

(define (global-of [d datum]) -> (ptr global-record))

(define (tag-of [d datum]) -> (ptr tag-record))

(define key-counter i64)
