;; cmd/sub/build
;; Building a program: its package and module map, the cache and the store, and the compiler, which sub runs with the
;; map and the grants a build has.
;; private to cmd
;; requires compiler/macros compiler/roots host/fs host/node/env host/events host/node/process std/package/manifest std/package/map std/package/path cmd/sub/common

(define (prepare! [file string]) -> string)

(define program-found i32)

(define program-manifest manifest)

(define program-dir string)

(define program-package string)

(define (find-package! [file string]) -> string)

(define (module-map! [text (ptr buffer)]))

;; supported! refuses a package that needs a newer substrate than this one.
(define (supported! [roots (slice package-root)]))

;; ---- the cache and the store ----

(define cache-dir string)

(define store-dir string)

(define (located [args (slice string)] [leading bool]) -> (slice string))

(define (locations! [project string]))

(define (cache!))

(define (target! [full string] [out string]) -> string)

(define (build! [file string] [out string] [strip bool] [next (fn (i32))]))

;; ---- the compiler ----

;; compile! runs the compiler on the program in full for command, build, expand or eval, into out with args, the module
;; map prepare! made and the given grants, and then next with its exit code.
(define (compile! [command string] [full string] [out string] [args (slice string)] [given (slice string)]
                  [next (fn (i32))]))

;; ---- grants ----

;; grants gives the compiler's grants for the program in full built into out: reading the roots, the cache and out, or
;; everything outside packages, and writing the cache and out.
(define (grants [full string] [out string]) -> (slice string))

(define (every-grant) -> (slice string))

;; Lists in grant flags write a comma as ",," (substrate/host/grants.mjs).
(define (granted! [b (ptr buffer)] [path string]))

;; ==== private ====

(define map-text buffer)

(define program-file string)

(define found-for string)

(define found-project string)

(define (package-of! [file string]) -> string)

(define packages (slice package-root))

(define (toolchain-file? [file string]) -> bool)

;; ---- the cache and the store ----

(define given-cache string)

(define given-packages string)

(define (location [given string] [variable string] [fallback string]) -> string)

;; ---- the compiler ----

(struct compiling [head event-head])

(define compiler-run compiling)

(define compiled-then (fn (i32)))

(define map-file string)

(define (argument! [b (ptr buffer)] [s string]))

(define (compiled [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))
