;; cmd/sub/doc
;; sub doc: the documentation of a package, or of a root of the toolchain (std, compiler, host, cmd), from its modules'
;; source: as text, each module as a header file would read and each directory as a list of its modules, or as HTML.
;; A package's modules are its files, the modules it exports public and the rest private; every module of a root of the
;; toolchain is public. A module's language describes its forms (std/doc/doc.sub).
;; requires compiler/macros compiler/expander compiler/reader host/fs std/package/manifest std/package/map std/package/path std/doc/doc std/doc/text std/doc/html cmd/sub/common cmd/sub/build

(define (doc-command! [args (slice string)]))

;; ==== private ====

(struct target [label string] [dir string] [modules datum] [public datum])

(define (target-of [given string]) -> target)

;; ---- targets ----

(define (toolchain-target [label string]) -> target)

;; modules-under gives every module of a root of the toolchain: each .sub file under its directory but its package.sub.
(define (modules-under [dir string] [rel string]) -> datum)

;; mapped! gives the roots of the module map of the package at dir, the package first, and gives the compiler the map.
(define (mapped! [dir string]) -> (slice package-root))

(define (package-target [r (ptr package-root)]) -> target)

(define (modules-in [dir string] [files (slice string)]) -> datum)

(define (sorted [l datum]) -> datum)

(define (identity-in [t (ptr target)] [rel datum]) -> datum)

(define (owner-of [t (ptr target)] [rel datum]) -> datum)

;; ---- writing ----

;; gathered gives a target's label; for each of its modules, its path, its documentation and the package it is private
;; to, if it is; and its directories' indexes.
(define (gathered [t (ptr target)]) -> datum)

(define (texts! [out string] [sets datum]))

(define (site! [out string] [sets datum]))

;; types-for gives each type a signature in a module may name, its own and those of the modules it requires, with the
;; module that defines it.
(define (types-for [d datum] [docs datum]) -> datum)

(define (directories [rels datum]) -> datum)

;; index-of gives a directory's name, the header of its main module (the root's is label.sub), its entries, and its
;; private modules' entries, from each module's path, name, header and the package it is private to: an entry is a
;; public module in it, or a directory in it, with the first sentence of a header (a directory's from its main module).
(define (index-of [t (ptr target)] [dir string] [heads datum]) -> datum)

(define (main-of [t (ptr target)] [dir string]) -> string)

(define (header-in [rel string] [heads datum]) -> datum)

(define (child-of [dir string] [rel string]) -> string)

(define (parent-in [rel string]) -> string)

(define (file! [out string] [path string] [content datum]))

;; ---- printing ----

;; printed! prints the module or directory a name means as the compiler finds it, in the package holding the current
;; directory, with its module map, or in the toolchain; or one item of the module.
(define (printed! [name string] [item string]))

;; project-of gives the package holding dir, or "" when none or a root of the toolchain does.
(define (project-of [dir string]) -> string)

(define (identity-named [name string] [from datum]) -> datum)

(define (head-of [s string]) -> string)

(define (package-in [roots (slice package-root)] [label string]) -> (ptr package-root))
