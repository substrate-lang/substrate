;; cmd/sub/doc
;; sub doc: the documentation of a package, or of a package of the toolchain (std, compiler, host, cmd), from its
;; modules' source: as text, each module as a header file would read and each directory as a list of its modules, or
;; as HTML. A module's language describes its forms (std/doc/doc.sub).
;; requires compiler/macros compiler/expander compiler/reader host/fs std/package/manifest std/package/path std/doc std/doc/text std/doc/html cmd/sub/common

(define (doc-command! [args (slice string)]))

;; ==== private ====

(struct target [label string] [dir string])

(define (target-of [given string]) -> target)

(define (toolchain? [s string]) -> bool)

;; ---- modules ----

(define (modules-of [t (ptr target)] [rel string]) -> datum)

(define (module-name [t (ptr target)] [rel string]) -> datum)

(define (source-text [file string]) -> string)

(define (forms-of [file string] [text string]) -> datum)

(define (documented [t (ptr target)] [rel string] [full bool]) -> datum)

;; named gives a module written by path, as a require or a language form writes it, by its name.
(define (named [t (ptr target)] [file string] [written datum]) -> datum)

;; ---- languages ----

(define descriptions datum)

(define (description [file string] [forms datum] [t (ptr target)]) -> datum)

(define (described [file string] [t (ptr target)]) -> datum)

(define (language-file [file string] [source datum] [t (ptr target)]) -> string)

;; ---- writing ----

;; gathered gives a package's label, its modules' documentation, and its directories' indexes.
(define (gathered [t (ptr target)]) -> datum)

(define (texts! [out string] [sets datum]))

(define (site! [out string] [sets datum]))

;; types-for gives each type a signature in a module may name, its own and those of the modules it requires, with the
;; module that defines it.
(define (types-for [d datum] [docs datum] [names datum]) -> datum)

(define (main-name [name datum] [names datum]) -> datum)

(define (directories [rels datum]) -> datum)

;; index-of gives a directory's name, the header of its main module, and its entries: each module in it, then each
;; directory in it, with the first sentence of a header (a directory's from its main module).
(define (index-of [t (ptr target)] [dir string] [docs datum]) -> datum)

(define (main-of [dir string]) -> string)

(define (header-in [rel string] [docs datum]) -> datum)

(define (child-of [dir string] [rel string]) -> string)

(define (parent-in [rel string]) -> string)

(define (file! [out string] [path string] [content datum]))

;; ---- printing ----

(define (printed! [name string] [item string]))
