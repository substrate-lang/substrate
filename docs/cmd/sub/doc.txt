;; cmd/sub/doc
;; sub doc: the documentation of packages and of the toolchain, substrate/, whose roots are compiler and host, written as
;; text or HTML (std/doc); with no arguments, the package holding the current directory and those its module map gives,
;; std among them, as HTML in its build/doc; or one module, directory or item of one, printed by the name the compiler
;; finds it by. A package's modules are its files, the modules it exports public and the rest private; every module of
;; the toolchain is public. Like build, it first fetches what the packages it reads depend on.
;; private to cmd
;; requires compiler/macros compiler/expander compiler/reader host/fs std/package/manifest std/package/map std/package/path std/doc/doc std/doc/root std/doc/text std/doc/html cmd/sub/common cmd/sub/build cmd/sub/fetch

(define (doc-command! [args (slice string)]))

;; ---- fetching ----

(define (doc-fetch-then! [args (slice string)] [next (fn ())]))

;; ==== private ====

(struct target [label string] [dir string] [modules datum] [public datum])

(define (package-docs!))

;; docs-of gives the documentation of a package's directory, or of each root of the toolchain's.
(define (docs-of [given string]) -> datum)

(define (doc-of [t target]) -> datum)

(define (toolchain-dir? [given string]) -> bool)

;; ---- targets ----

(define (toolchain-target [label string]) -> target)

;; modules-under gives every module of a root of the toolchain: each .sub file under its directory.
(define (modules-under [dir string] [rel string]) -> datum)

;; mapped! gives the roots of the module map of the package at dir, the package first, and gives the compiler the map.
(define (mapped! [dir string]) -> (slice package-root))

(define (package-target [r (ptr package-root)]) -> target)

(define (modules-in [dir string] [files (slice string)]) -> datum)

(define (sorted [l datum]) -> datum)

(define (files! [out string] [files datum]))

(define (file! [out string] [path string] [content datum]))

;; ---- fetching ----

(define unfetched datum)

(define after (fn ()))

(define (fetched!))

;; ---- printing ----

;; printed! prints the module or directory a name means as the compiler finds it, in the package holding the current
;; directory, with its module map, or outside packages; or one item of the module.
(define (printed! [name string] [item string]))

(define (identity-named [name string] [from datum]) -> datum)

(define (head-of [s string]) -> string)
