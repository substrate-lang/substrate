;; sqlite/vfs
;; SQLite's files, on the host's (host/fs.sub): the VFS sqlite3_os_init registers as SQLite starts, with locks between the
;; connections of one instance of the program.
;; requires host/fs host/modules/clock host/modules/random sqlite/sqlite3.wasm

(define (c-string [p (ptr u8)]) -> string)

(define (c-copy [s string]) -> (ptr u8))

;; ---- the VFS ----

(define (vfs-start!) -> i32)

;; ==== private ====

(const ok i32 0)

(const busy i32 5)

(const not-found i32 12)

(const cannot-open i32 14)

(const read-error i32 266)

(const short-read i32 522)

(const write-error i32 778)

(const sync-error i32 1034)

(const truncate-error i32 1546)

(const size-error i32 1802)

(const delete-error i32 2570)

(const close-error i32 4106)

(const deleted-missing i32 5898)

(const open-read-only i32 1)

;; sqlite3_io_methods, version 1, and sqlite3_vfs, version 2.
(struct io-methods [version i32] [close i32] [read i32] [write i32] [truncate i32] [sync i32] [size i32] [lock i32]
  [unlock i32] [check-reserved-lock i32] [file-control i32] [sector-size i32] [device-characteristics i32])

(struct vfs [version i32] [file-size i32] [max-pathname i32] [next i32] [name i32] [app-data i32] [open i32]
  [delete i32] [access i32] [full-pathname i32] [dl-open i32] [dl-error i32] [dl-sym i32] [dl-close i32]
  [randomness i32] [sleep i32] [current-time i32] [last-error i32] [current-time-int64 i32])

;; A path's lock, which its open files share: the strongest any holds, and how many hold a shared lock or more.
(struct lock-node [next (ptr lock-node)] [path string] [lock i32] [shared i32] [files i32])

(struct vfs-file [methods (ptr io-methods)] [handle u32] [lock i32] [node (ptr lock-node)] [delete bool] [path string])

(define methods io-methods)

(define the-vfs vfs)

(define nodes (ptr lock-node))

(define (node-of [path string]) -> (ptr lock-node))

(define (node-done! [n (ptr lock-node)]))

;; ---- files ----

(define (file-close [f (ptr vfs-file)]) -> i32)

(define (file-read [f (ptr vfs-file)] [buf (ptr u8)] [n i32] [at i64]) -> i32)

(define (file-write [f (ptr vfs-file)] [buf (ptr u8)] [n i32] [at i64]) -> i32)

(define (file-truncate [f (ptr vfs-file)] [n i64]) -> i32)

(define (file-sync [f (ptr vfs-file)] [flags i32]) -> i32)

(define (file-size [f (ptr vfs-file)] [out (ptr i64)]) -> i32)

;; The locks of unixLock and posixUnlock, within the program.
(define (file-lock [f (ptr vfs-file)] [level i32]) -> i32)

(define (file-unlock [f (ptr vfs-file)] [level i32]) -> i32)

(define (file-check-reserved-lock [f (ptr vfs-file)] [out (ptr i32)]) -> i32)

(define (file-control [f (ptr vfs-file)] [op i32] [arg i32]) -> i32)

(define (file-sector-size [f (ptr vfs-file)]) -> i32)

(define (file-device-characteristics [f (ptr vfs-file)]) -> i32)

;; ---- the VFS ----

(define (vfs-open [v i32] [name (ptr u8)] [f (ptr vfs-file)] [flags i32] [out (ptr i32)]) -> i32)

(define (vfs-delete [v i32] [name (ptr u8)] [sync-directory i32]) -> i32)

;; A file exists for SQLite when it is not empty.
(define (vfs-access [v i32] [name (ptr u8)] [flags i32] [out (ptr i32)]) -> i32)

(define (vfs-full-pathname [v i32] [name (ptr u8)] [room i32] [out (ptr u8)]) -> i32)

(define (vfs-randomness [v i32] [n i32] [out (ptr u8)]) -> i32)

(define (vfs-sleep [v i32] [microseconds i32]) -> i32)

;; The current time as a Julian day number, and in milliseconds since the Julian epoch.
(define (vfs-current-time [v i32] [out (ptr f64)]) -> i32)

(define (vfs-current-time-int64 [v i32] [out (ptr i64)]) -> i32)

(define (vfs-last-error [v i32] [n i32] [out (ptr u8)]) -> i32)
