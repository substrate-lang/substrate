;; sqlite/sqlite
;; SQLite, from sqlite3.wasm (build.sh), on the host's files (vfs.sub), after SQLite's C calls. A call that fails
;; returns -1, or 0 for a database or statement, and stores SQLite's message at err. A row's text and blobs are
;; SQLite's until its statement steps again, is reset or is finalized.
;; requires sqlite/vfs sqlite/sqlite3.wasm

(struct sqlite)

(struct sqlite-statement)

(const sqlite-row i32 100)

(const sqlite-done i32 101)

(const sqlite-integer i32 1)

(const sqlite-float i32 2)

(const sqlite-text i32 3)

(const sqlite-blob i32 4)

(const sqlite-null i32 5)

(define (sqlite-open [path string] [err (ptr string)]) -> (ptr sqlite))

(define (sqlite-close [db (ptr sqlite)]))

;; sql is copied once with a 0 after it, as SQLite copies text without one each time it prepares a statement.
(define (sqlite-exec [db (ptr sqlite)] [sql string] [err (ptr string)]) -> i32)

(define (sqlite-prepare [db (ptr sqlite)] [sql string] [err (ptr string)]) -> (ptr sqlite-statement))

(define (sqlite-step [st (ptr sqlite-statement)] [err (ptr string)]) -> i32)

(define (sqlite-reset [st (ptr sqlite-statement)]))

(define (sqlite-finalize [st (ptr sqlite-statement)]))

;; ---- binding, from 1 ----

(define (sqlite-bind-null [st (ptr sqlite-statement)] [i i32]) -> i32)

(define (sqlite-bind-i64 [st (ptr sqlite-statement)] [i i32] [v i64]) -> i32)

(define (sqlite-bind-f64 [st (ptr sqlite-statement)] [i i32] [v f64]) -> i32)

(define (sqlite-bind-text [st (ptr sqlite-statement)] [i i32] [s string]) -> i32)

(define (sqlite-bind-blob [st (ptr sqlite-statement)] [i i32] [b (slice u8)]) -> i32)

;; ---- columns, from 0 ----

(define (sqlite-column-count [st (ptr sqlite-statement)]) -> i32)

(define (sqlite-column-type [st (ptr sqlite-statement)] [i i32]) -> i32)

(define (sqlite-column-name [st (ptr sqlite-statement)] [i i32]) -> string)

(define (sqlite-column-i64 [st (ptr sqlite-statement)] [i i32]) -> i64)

(define (sqlite-column-f64 [st (ptr sqlite-statement)] [i i32]) -> f64)

(define (sqlite-column-text [st (ptr sqlite-statement)] [i i32]) -> string)

(define (sqlite-column-blob [st (ptr sqlite-statement)] [i i32]) -> (slice u8))

;; ---- the database ----

(define (sqlite-changes [db (ptr sqlite)]) -> i64)

(define (sqlite-last-insert-rowid [db (ptr sqlite)]) -> i64)

(define (sqlite-version) -> string)

;; ==== private ====

(const read-write-create i32 6)

(const transient i32 -1)

;; Empty text and blobs are bound at this address, as SQLite takes address 0 as NULL.
(define empty (array u8 1))

(define (failure [db i32] [err (ptr string)]) -> i32)

;; ---- binding, from 1 ----

(define (bound [r i32]) -> i32)

(define (bytes-at [b (slice u8)]) -> i32)
