;; std/compress/inflate
;; DEFLATE decoding (RFC 1951) of whole buffers, with lookup tables as zlib and libdeflate decode. inflate! decodes the
;; data at the start of in, adding at most most bytes to out, sets used to the bytes it took, and returns 0 or a negative
;; code whose message, zlib's, inflate-message gives.

(define (inflate-message [code i32]) -> string)

;; ---- decoding ----

(define (inflate! [out (ptr buffer)] [in (slice u8)] [most u32] [used (ptr u32)]) -> i32)

;; ==== private ====

;; Table entries hold a code's length in bits, an op and a value. Op 0 is a literal, 16 plus a count of extra bits a
;; base length or distance, 32 the end of a block, 64 plus a width a link to the table of that many bits at the value,
;; and 128 a code that is not allowed. A code longer than a table's root bits is found through a link.
(const op-base u32 16)

(const op-end u32 32)

(const op-link u32 64)

(const op-bad u32 128)

(const end-of-data i32 -1)

(const bad-block-type i32 -2)

(const bad-stored-lengths i32 -3)

(const too-many-symbols i32 -4)

(const bad-code-lengths i32 -5)

(const bad-repeat i32 -6)

(const no-end-code i32 -7)

(const bad-lengths-set i32 -8)

(const bad-distances-set i32 -9)

(const bad-length-code i32 -10)

(const bad-distance-code i32 -11)

(const too-far-back i32 -12)

(const too-large i32 -13)

(define length-base (array u32 29))

(define length-extra (array u32 29))

(define distance-base (array u32 30))

(define distance-extra (array u32 30))

;; The order the lengths of the code length code come in.
(define code-length-order (array u32 19))

;; ---- tables ----

;; Kinds of table: code lengths, literals and lengths, and distances.
(const codes i32 0)

(const lengths i32 1)

(const distances i32 2)

(define (entry-of [kind i32] [s u32]) -> u32)

(define (reversed [code u32] [n u32]) -> u32)

;; The width of the table each root entry links to, and where that table starts.
(define widths (array u8 1024))

(define starts (array u32 1024))

;; build! fills table, of room entries, for the codes of the n lengths: an entry for each of the 2^root values of the
;; next root bits, then the tables longer codes link to. It returns #f when the lengths are too many for a code, or too
;; few for anything but a lone code of one bit, which zlib allows for distances.
(define (build! [table (ptr u32)] [room u32] [root u32] [lens (ptr u8)] [n u32] [kind i32]) -> bool)

(define fixed-lengths (array u32 1024))

(define fixed-distances (array u32 32))

(define fixed bool)

(define (fixed!))

;; ---- decoding ----

(define code-table (array u32 128))

(define length-table (array u32 2048))

(define distance-table (array u32 1024))

;; The bits not yet used, count of them, the bytes still to read, and the output with where this stream's starts and
;; how far it may go.
(struct reader [bits u64] [count u32] [at (ptr u8)] [end (ptr u8)] [out (ptr buffer)] [start u32] [most u32])

;; refill! has the reader hold at least 56 bits, or every bit left. With 8 bytes to read it loads them at once, keeping
;; whole bytes, as libdeflate does: the bits above count are the next bytes' own, so loading them again changes nothing.
(define (refill! [r (ptr reader)]))

;; take! uses n bits, which the reader must hold, and gives them.
(define (take! [r (ptr reader)] [n u32]))

;; The entry for the code at the start of the bits, through a link if it has one.
(define (entry [table (ptr u32)] [root u32] [bits u64]))

;; room! has the output hold n more bytes, and 8 to spare, and returns #f when that passes most.
(define (room! [r (ptr reader)] [n u32]))

(define (stored! [r (ptr reader)]) -> i32)

;; header! reads a dynamic block's codes into the length and distance tables.
(define (header! [r (ptr reader)]) -> i32)

(define (codes! [r (ptr reader)] [lt (ptr u32)] [lroot u32] [dt (ptr u32)] [droot u32]) -> i32)
