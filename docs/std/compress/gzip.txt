;; std/compress/gzip
;; The gzip format (RFC 1952) around DEFLATE data, and its CRC-32. gunzip! decodes whole gzip data, each member in
;; turn, adding at most most bytes to out, and returns 0 or a negative code whose message gzip-message gives.
;; requires std/compress/inflate std/compress/deflate

;; crc32 continues the CRC-32 crc, 0 at first, over the bytes.
(define (crc32 [crc u32] [b (slice u8)]) -> u32)

;; ---- writing ----

;; gzip-start! writes a member's header, with name as its file name unless empty; gzip-end! its trailer, the CRC-32 and
;; length of what it holds.
(define (gzip-start! [out (ptr buffer)] [name string]))

(define (gzip-end! [out (ptr buffer)] [crc u32] [size u32]))

;; gzip! writes in as one member, compressed by d from a fresh start.
(define (gzip! [d (ptr deflater)] [out (ptr buffer)] [in (slice u8)] [name string]))

;; ---- reading ----

(define (gzip-message [code i32]) -> string)

(define (gunzip! [out (ptr buffer)] [in (slice u8)] [most u32]) -> i32)

;; ==== private ====

;; CRC-32 with eight tables, slicing eight bytes at a time.
(define tables (array u32 2048))

(define tabled bool)

(define (tables!))

;; ---- reading ----

(const bad-magic i32 -101)

(const bad-method i32 -102)

(const bad-flags i32 -103)

(const bad-header-check i32 -104)

(const bad-data-check i32 -105)

(const bad-length-check i32 -106)

(const short-header i32 -107)

(define (le32 [p (ptr u8)]) -> u32)

;; The length of the member's header at the start of b, or a negative code.
(define (header-length [b (slice u8)]) -> i32)
