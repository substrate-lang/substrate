;; std/compress/deflate
;; DEFLATE compression (RFC 1951) as a stream, after zlib: a 32 KB window, hash chains of 3-byte strings, greedy matching
;; at levels 1 to 3 and lazy at 4 to 9 with zlib's settings, and each block stored, fixed or with its own codes,
;; whichever is shortest. deflate! adds in's compressed form to out: deflate-none may hold some back, deflate-sync sends
;; everything and ends at a whole byte, deflate-finish ends the stream. deflater-reset! starts another stream, and
;; deflate-alone! compresses bytes as if nothing came before, ending as a sync, so one deflater serves many streams.

(const deflate-none i32 0)

(const deflate-sync i32 1)

(const deflate-finish i32 2)

(struct deflater [level i32] [good u32] [lazy u32] [nice u32] [chain u32]
  [window (ptr u8)] [head (ptr u16)] [prev (ptr u16)] [syms (ptr u32)] [count u32]
  [strstart u32] [lookahead u32] [block-start i32] [floor u32]
  [match-length u32] [match-start u32] [prev-length u32] [prev-match u32] [available bool]
  [bits u64] [nbits u32] [lit-freq (array u32 288)] [dist-freq (array u32 30)])

;; ---- the window ----

(define (deflater-init! [d (ptr deflater)] [level i32]))

(define (deflater-free! [d (ptr deflater)]))

;; deflater-reset! starts a stream after whatever the deflater did before, which no match may reach back into: past the
;; first half of the window it clears the hash heads and starts again at its start, as zlib's deflateReset does.
(define (deflater-reset! [d (ptr deflater)]))

;; ---- matching ----

(define (deflate! [d (ptr deflater)] [out (ptr buffer)] [in (slice u8)] [flush i32]))

(define (deflate-alone! [d (ptr deflater)] [out (ptr buffer)] [in (slice u8)]))

;; ==== private ====

(const wsize u32 32768)

(const wmask u32 32767)

(const hash-bits u32 15)

(const hash-size u32 32768)

(const min-match u32 3)

(const max-match u32 258)

(const min-lookahead u32 262)

(const max-dist u32 32506)

(const too-far u32 4096)

(const most-symbols u32 16383)

;; zlib's settings for each level: good, past which a search is cut to a quarter; lazy, which ends lazy searching (at
;; levels 1 to 3, the longest match whose strings are hashed); nice, which ends a search; and chain, the most links.
(define settings (array u32 40))

;; ---- tables ----

(define length-base (array u32 29))

(define length-extra (array u32 29))

(define distance-base (array u32 30))

(define distance-extra (array u32 30))

(define code-length-order (array u32 19))

;; The length code of each length less 3, the distance code of each distance less 1 below 256 and then of each 128th,
;; and the fixed codes, bit-reversed as they are sent, with their lengths.
(define length-code (array u8 256))

(define distance-code (array u8 512))

(define fixed-codes (array u32 288))

(define fixed-lens (array u8 288))

(define tabled bool)

(define (reversed [code u32] [n u32]) -> u32)

;; codes! gives the n lengths their canonical codes, reversed.
(define (codes! [lens (ptr u8)] [n u32] [out (ptr u32)]))

(define (tables!))

(define (dcode [d u32]))

;; ---- codes ----

;; huffman! gives the n symbols code lengths of at most most bits: a Huffman code from two queues, leaves and joined
;; nodes; lengths past most are cut and the code made whole again as miniz does; the lengths go out by frequency.
(define (huffman! [freq (ptr u32)] [n u32] [most u32] [lens (ptr u8)]))

;; ---- the window ----

(define (clear-freqs! [d (ptr deflater)]))

(define (slide! [d (ptr deflater)]))

;; fill! moves bytes of in from at into the window until it holds a match's look ahead, sliding it down as it fills.
(define (fill! [d (ptr deflater)] [in (slice u8)] [at (ptr u32)]))

;; insert! adds the string at pos to its hash chain and gives the chain's last string before it.
(define (insert! [d (ptr deflater)] [pos u32]))

;; The latest position a match may not start at or before.
(define (limit [d (ptr deflater)]))

;; longest! follows the chain from cur for the longest match at strstart.
(define (longest! [d (ptr deflater)] [cur u32]) -> u32)

;; ---- blocks ----

(define (literal! [d (ptr deflater)] [c u8]))

(define (match! [d (ptr deflater)] [distance u32] [len u32]))

;; put! sends the n low bits of v, n at most 32, into out, which has room for them.
(define (put! [d (ptr deflater)] [out (ptr buffer)] [v u32] [n u32]))

;; align! sends the bits left, filling the last byte with zeros.
(define (align! [d (ptr deflater)] [out (ptr buffer)]))

;; stored! sends bytes as stored blocks, the last of them final when last is set.
(define (stored! [d (ptr deflater)] [out (ptr buffer)] [bytes (slice u8)] [last bool]))

;; The cost in bits of the block's symbols with the codes of these lengths.
(define (cost [d (ptr deflater)] [llens (ptr u8)] [dlens (ptr u8)]) -> u64)

(define (send! [d (ptr deflater)] [out (ptr buffer)] [lcodes (ptr u32)] [llens (ptr u8)] [dcodes (ptr u32)]
               [dlens (ptr u8)]))

;; at-least-two! gives a code two symbols at least, for decoders that need them.
(define (at-least-two! [freq (ptr u32)] [n u32]))

;; block! sends the symbols so far as one block, and the window's bytes since the block began when stored.
(define (block! [d (ptr deflater)] [out (ptr buffer)] [last bool]))

;; ---- matching ----

;; greedy! takes a match wherever it finds one, and returns #t when it waits for more of the stream.
(define (greedy! [d (ptr deflater)] [out (ptr buffer)] [in (slice u8)] [at (ptr u32)] [flush i32]) -> bool)

;; lazy! takes a match only when the next position has none longer.
(define (lazy! [d (ptr deflater)] [out (ptr buffer)] [in (slice u8)] [at (ptr u32)] [flush i32]) -> bool)
