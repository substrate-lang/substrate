;; std/encoding/json
;; JSON (RFC 8259): values after cJSON's, in an arena; a token reader after Go's Decoder.Token; writing as JavaScript's
;; JSON.stringify writes, but -0 as -0; and the json form, a type's writer and reader after serde's derive.
;; requires std/encoding/utf8 compiler/macros std/meta/derive

(variant json-value
  [json-null]
  [json-boolean [value bool]]
  [json-number [value f64] [text string]]
  [json-string [text string]]
  [json-array [items (slice json-value)] [room u32]]
  [json-object [members (slice json-member)] [room u32]])

(struct json-member [key string] [value json-value])

(struct json-error [message string] [offset u32] [line u32] [column u32])

;; ---- the token reader ----

(const json-token-error u32 0)

(const json-token-end u32 1)

(const json-token-start-array u32 2)

(const json-token-end-array u32 3)

(const json-token-start-object u32 4)

(const json-token-end-object u32 5)

(const json-token-key u32 6)

(const json-token-string u32 7)

(const json-token-number u32 8)

(const json-token-true u32 9)

(const json-token-false u32 10)

(const json-token-null u32 11)

(struct json-reader [text string] [at u32] [state u32] [depth u32] [kinds (array u8 1000)] [token string] [number f64]
        [scratch buffer] [error json-error])

(define (json-reader-start! [r (ptr json-reader)] [text string]))

(define (json-reader-free! [r (ptr json-reader)]))

(define (json-next! [r (ptr json-reader)]) -> u32)

(define (json-skip! [r (ptr json-reader)]) -> bool)

(define (json-push! [a (ptr arena)] [array (ptr json-value)] [v json-value]) -> bool)
