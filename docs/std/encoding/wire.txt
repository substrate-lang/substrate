;; std/encoding/wire
;; Wire messages, encoded in the postcard 1.x wire format (postcard.jamesmunns.com/wire-format).
;; requires std/encoding/utf8 compiler/macros std/meta/derive

(struct wire-error [message string] [offset u32])

;; ---- the wire form ----

(define-syntax wire)

;; ==== private ====

(struct wire-reader [bytes (slice u8)] [at u32] [error wire-error])

;; ---- writing ----

(define (put-varint! [b (ptr buffer)] [n u64]))

(define (put-zigzag! [b (ptr buffer)] [n i64]))

(define (put-f64! [b (ptr buffer)] [x f64]))

(define (put-text! [b (ptr buffer)] [s string]))

;; ---- reading ----

(define (start! [r (ptr wire-reader)] [bytes (slice u8)]))

(define (fail! [r (ptr wire-reader)] [message string]) -> bool)

(define (refuse! [r (ptr wire-reader)] [at u32] [message string]) -> bool)

(define (left [r (ptr wire-reader)]) -> u32)

(define (end! [r (ptr wire-reader)]) -> bool)

(define (read-failed [r (ptr wire-reader)] [err (ptr wire-error)]) -> bool)

(define (take-byte! [r (ptr wire-reader)] [to (ptr u8)]) -> bool)

(define (take-raw! [r (ptr wire-reader)] [to (ptr void)] [n u32]) -> bool)

(define (take-flag! [r (ptr wire-reader)] [to (ptr bool)] [message string]) -> bool)

;; take-varint! reads a varint no longer and no larger than a type of the given bits allows.
(define (take-varint! [r (ptr wire-reader)] [bits u32] [to (ptr u64)]) -> bool)

(define (unzigzag [n u64]) -> i64)

(define (take-count! [r (ptr wire-reader)] [to (ptr u32)]) -> bool)

(define (take-text! [r (ptr wire-reader)] [a (ptr arena)] [to (ptr string)]) -> bool)

;; ---- the wire form ----

(define (literal [d datum]) -> datum)

(define (wire-form [s datum]) -> datum)

(define (definitions [t datum] [info datum]) -> datum)

(define (struct-code [t datum] [fields datum]) -> datum)

(define (variant-code [t datum] [cases datum]) -> datum)

(define (entry [t datum]) -> datum)

(define (bits [t datum]) -> datum)

;; Code writing the value at place p, of the field where.
(define (write-code [type datum] [p datum] [where datum]) -> datum)

;; Code reading into place p, of the field where.
(define (read-code [type datum] [p datum] [where datum]) -> datum)
