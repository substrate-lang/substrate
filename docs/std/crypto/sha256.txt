;; std/crypto/sha256
;; SHA-256 (FIPS 180-4), over bytes added in pieces or given at once.

(struct sha256 [state (array u32 8)] [block (array u8 64)] [filled u32] [length u64])

(define (sha256-init! [s (ptr sha256)]))

(define (sha256-add! [s (ptr sha256)] [b (slice u8)]))

;; into takes the 32 bytes of the digest.
(define (sha256-end! [s (ptr sha256)] [into (slice u8)]))

(define (sha256-of [b (slice u8)] [into (slice u8)]))

;; ==== private ====

(define k (array u32 64))

(define (rotr [x u32] [n u32]))

(define (word [p (ptr u8)]))

(define (compress! [s (ptr sha256)] [p (ptr u8)]))
