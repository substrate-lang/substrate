;; std/crypto/hmac
;; HMAC (RFC 2104) with SHA-256, and what is built on it: HKDF (RFC 5869), which derives keys from a secret, and
;; PBKDF2 (RFC 8018), which derives them from a password.
;; requires std/crypto/sha256

(struct hmac [inner sha256] [outer sha256])

(define (hmac-init! [m (ptr hmac)] [key (slice u8)]))

(define (hmac-add! [m (ptr hmac)] [b (slice u8)]))

;; into takes the 32 bytes of the code.
(define (hmac-end! [m (ptr hmac)] [into (slice u8)]))

(define (hmac-of [key (slice u8)] [b (slice u8)] [into (slice u8)]))

;; hmac-equal? takes time that depends only on the lengths.
(define (hmac-equal? [a (slice u8)] [b (slice u8)]) -> bool)

;; hkdf! fills out, at most 8160 bytes, with keys from the secret, the salt and info, which names their use.
(define (hkdf! [secret (slice u8)] [salt (slice u8)] [info (slice u8)] [out (slice u8)]))

(define (pbkdf2! [password (slice u8)] [salt (slice u8)] [iterations u32] [out (slice u8)]))
