;; std/crypto/sha1
;; SHA-1 (RFC 3174), over bytes added in pieces or given at once.

(struct sha1 [h (array u32 5)] [block (array u8 64)] [filled u32] [length u64])

(define (sha1-init! [s (ptr sha1)]))

(define (sha1-add! [s (ptr sha1)] [b (slice u8)]))

(define (sha1-end! [s (ptr sha1)] [into (slice u8)]))

(define (sha1-of [b (slice u8)] [into (slice u8)]))

;; ==== private ====

(define (rotl [x u32] [k u32]))

(define (word [p (ptr u8)]) -> u32)

(define (compress! [s (ptr sha1)] [p (ptr u8)]))
