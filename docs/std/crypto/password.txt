;; std/crypto/password
;; Passwords stored as PHC strings of PBKDF2-SHA256, $pbkdf2-sha256$i=ITERATIONS$SALT$HASH in Base64 without padding,
;; worked out in a worker. password-hash! gives done a new password's string in result, after the job's iterations,
;; 600000 when 0. password-verify! tells done in ok whether a password matches a stored string; one that cannot be read
;; takes as long and does not match. Jobs run in order; if the worker ends, each waiting job ends without a result.
;; requires host/events host/modules/random host/worker std/crypto/hmac std/encoding/base64

(struct password-job [user (ptr void)] [iterations u32] [result buffer] [ok bool] [verify bool] [readable bool]
  [password buffer] [salt (array u8 16)] [expected buffer] [next (ptr password-job)] [done (fn ((ptr password-job)))])

;; ---- the program's side ----

(define (password-hash! [j (ptr password-job)] [password string] [done (fn ((ptr password-job)))]))

(define (password-verify! [j (ptr password-job)] [password string] [stored string] [done (fn ((ptr password-job)))]))

(define (password-job-free! [j (ptr password-job)]))

;; ==== private ====

(const default-iterations u32 600000)

(const most-iterations u64 100000000)

(define first (ptr password-job))

(define last (ptr password-job))

(define running u32)

(define listener event-head)

;; ---- the worker's side ----

;; A request is the iterations in 4 bytes, little-endian, the salt's length in a byte, the salt and the password; the
;; answer is the 32-byte key.
(define (requested [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))

(define (password-worker))

;; ---- the program's side ----

(define (wipe! [b (ptr buffer)]))

(define (send-first!))

(define (start! [j (ptr password-job)] [done (fn ((ptr password-job)))]))

(define (unpadded! [b (ptr buffer)] [bytes (slice u8)]))

(define (finish! [key (slice u8)]))

(define (failed!))

(define (answered [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))

;; read! takes the iterations, salt and hash from a stored string, or returns #f.
(define (read! [j (ptr password-job)] [stored string]) -> bool)

(define (padded-decode! [b (ptr buffer)] [text string]) -> bool)
