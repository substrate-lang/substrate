;; std/package/map
;; The module map the compiler takes for a program in a package: a root for the package and for each package its
;; dependencies reach, with the names each uses, and each package's exports and files for sub's module check and the
;; substrate it needs; and std, a name every module has, the std a package depends on or else the package at std. A
;; pinned dependency's hash is checked, and two packages with one name, and a package that depends on itself, are
;; refused.
;; requires std/package/manifest std/package/hash std/package/files host/fs std/package/path std/package/store

(struct package-root [name string] [dir string] [file string] [substrate string] [exports (slice string)]
                     [files (slice string)])

(define (package-map [a (ptr arena)] [dir string] [file string] [m (ptr manifest)] [std string] [store string]
                     [into (ptr buffer)] [roots (ptr buffer)] [err (ptr buffer)]) -> bool)
