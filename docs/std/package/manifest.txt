;; std/package/manifest
;; Manifests: a package's package.sub, or a program's first form, (package name (version "major.minor.patch")
;; (exports module ...) (dependencies (name (path "directory") (hash "sha256:hex")) (name (url "address") (hash
;; "sha256:hex")) ...)), the exports, the dependencies and a path's hash being optional.
;; requires std/package/read host/fs std/package/path

(struct dependency [name string] [path string] [url string] [hash string] [line u32] [column u32])

(struct manifest [name string] [version string] [major u32] [exports (slice string)] [dependencies (slice dependency)])

(define (manifest-read [a (ptr arena)] [file string] [m (ptr manifest)] [err (ptr buffer)]) -> bool)

(define (manifest-of-program [a (ptr arena)] [file string] [m (ptr manifest)] [err (ptr buffer)]) -> i32)

;; manifest-find finds the manifest of the package that holds the program in file, given from /: the program's own,
;; giving 1, or the nearest package.sub above it, giving 2, with the package's directory added to dir. It gives 0 when
;; there is none, and -1 with err saying why when the manifest is wrong.
(define (manifest-find [a (ptr arena)] [file string] [m (ptr manifest)] [dir (ptr buffer)] [err (ptr buffer)]) -> i32)

(define (package-holding [a (ptr arena)] [dir string]) -> string)

(define (package-name? [s string]) -> bool)

(define (full-hash? [s string]) -> bool)

;; ==== private ====

(define (read-forms [a (ptr arena)] [file string] [first bool] [into (ptr (slice data))] [e (ptr data-error)]
                    [err (ptr buffer)]) -> bool)

(define (failed! [err (ptr buffer)] [file string] [line u32] [column u32] [message string]) -> bool)

(define (failed-at! [err (ptr buffer)] [file string] [d (ptr data)] [message string]) -> bool)

(define (symbol-of [d (ptr data)]) -> string)

(define (string-of [d (ptr data)] [into (ptr string)]) -> bool)

(define (items-of [d (ptr data)]) -> (slice data))

(define (list-of? [d (ptr data)] [head string] [n u32]) -> bool)

(define (version-major [s string]) -> i64)

(define (manifest-of [file string] [f (ptr data)] [a (ptr arena)] [m (ptr manifest)] [err (ptr buffer)]) -> bool)

(define (exports-of [file string] [c (ptr data)] [a (ptr arena)] [m (ptr manifest)] [err (ptr buffer)]) -> bool)

(define (module-name? [s string]) -> bool)

(define (dependencies-of [file string] [c (ptr data)] [a (ptr arena)] [m (ptr manifest)] [err (ptr buffer)]) -> bool)

(define dependency-form string)
