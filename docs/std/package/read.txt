;; std/package/read
;; Data in Substrate's syntax, as package.sub holds it: lists in ( ) or [ ], symbols and strings, with the comments and
;; string escapes of Substrate's reader, read into an arena.

(variant data
  [data-list [items (slice data)] [line u32] [column u32]]
  [data-symbol [text string] [line u32] [column u32]]
  [data-string [text string] [line u32] [column u32]])

(struct data-error [message string] [line u32] [column u32])

(define (data-read [a (ptr arena)] [text string] [first bool] [into (ptr (slice data))] [err (ptr data-error)]) -> bool)

(define (data-head [a (ptr arena)] [text string]) -> string)

;; ==== private ====

(struct data-reader [text string] [at u32] [line u32] [column u32] [arena (ptr arena)] [error (ptr data-error)])

(define (copied [a (ptr arena)] [b (slice u8)]) -> (ptr u8))

(define (here [r (ptr data-reader)]) -> u32)

(define (more? [r (ptr data-reader)]) -> bool)

(define (advance! [r (ptr data-reader)]))

(define (after [r (ptr data-reader)]) -> u32)

(define (skip-space! [r (ptr data-reader)]) -> bool)

(define (skip-block! [r (ptr data-reader)]) -> bool)

(define (delimiter? [c u32]) -> bool)

(define (failed! [r (ptr data-reader)] [message string] [line u32] [column u32]) -> bool)

(define (read-one! [r (ptr data-reader)] [into (ptr data)]) -> bool)

(define (read-list! [r (ptr data-reader)] [close u32] [line u32] [column u32] [into (ptr data)]) -> bool)

(define (read-string! [r (ptr data-reader)] [line u32] [column u32] [into (ptr data)]) -> bool)

(define (hex-escape! [r (ptr data-reader)] [b (ptr buffer)]) -> bool)

(define (hex-value [c u32]) -> i32)
