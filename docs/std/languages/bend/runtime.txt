;; std/languages/bend/runtime
;; The runtime of Bend's compiled code: nodes, reference counting and printing.
;; A term: bit 63 shared; bits 56-62 tag (0 packed, 1 Nat, 2 node, 3 closure); bits 0-31 address;
;; bits 32-55 a node's constructor (low 12) and field count (high 12), or a closure's table index.
;; requires std/languages/nodes

(define (node [n u32]))

(define (free-node [a i32] [n u32]))

(define (node-pop [k u32]))

(define (node-push [a i32] [k u32]))

(define (static [n u32]) -> i32)

(define (live) -> i32)

;; A shared term points at a one-word redirect: its node's address above, its owner count below.
(define (share [t u64] [k u32]) -> u64)

(define (take [t u64] [n u32]) -> i32)

(define (node-of [t u64]) -> i32)

(define (drop [t u64]))

(define (word-u32 [w u64]) -> u64)

(define (u32-word [t u64]) -> u64)

(define (write-u32 [t u64]))

;; As Bend prints a Nat: Zero, or (nat n).
(define (write-nat [t u64]))

;; ==== private ====

(define (shared? [t u64]) -> bool)

(define (tag-of [t u64]) -> u32)

(define (loc-of [t u64]) -> (ptr u64))

(define (trivial? [t u64]) -> bool)

(define stack (ptr u64))

(define room u32)

(define (push! [top u32] [t u64]))
