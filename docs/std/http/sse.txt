;; std/http/sse
;; Server-sent events, the HTML standard's event streams. sse-accept! answers a request with a stream, and sse-send!
;; sends each event on it. A comment goes out every beat milliseconds, 15 s when 0. on-closed is called if the client
;; leaves before sse-close! ends the stream, or after the server ends it as it shuts down. Once accepted, the request's
;; user is the stream. compress, set before sse-accept!, has the events gzipped, each alone, when the server compresses.
;; requires host/events host/modules/timer std/http/http std/http/server

(struct sse-stream [head event-head] [request (ptr http-request)] [user (ptr void)] [beat f64] [compress bool]
  [timer u32] [out buffer] [hook http-hook] [on-closed (fn ((ptr sse-stream)))] [on-drained (fn ((ptr sse-stream)))])

;; sse-accept! answers with an event stream and returns #t, or #f when the request is HEAD or the client is gone.
;; X-Accel-Buffering: no keeps nginx from holding events back.
(define (sse-accept! [s (ptr sse-stream)] [r (ptr http-request)]) -> bool)

;; sse-send! sends an event of type event, "message" when empty, with id unless empty, and data. It returns what
;; http-write! does, or #f once the stream is closed. An event or id with a line break or NUL traps.
(define (sse-send! [s (ptr sse-stream)] [event string] [id string] [data string]) -> bool)

;; sse-retry! tells the client how many milliseconds to wait before it reconnects.
(define (sse-retry! [s (ptr sse-stream)] [ms u32]) -> bool)

;; sse-close! ends the stream without calling on-closed.
(define (sse-close! [s (ptr sse-stream)]))

;; ==== private ====

(const default-beat f64 15000.0)

(define (arm! [s (ptr sse-stream)]))

(define (stop! [s (ptr sse-stream)]))

(define (stopping [tag (ptr void)] [forced bool]))

(define (gone [r (ptr http-request)]))

(define (drained [r (ptr http-request)]))

(define (beat-heard [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))

(define (field? [s string]) -> bool)

(define (field! [b (ptr buffer)] [name string] [value string]))

;; lines! adds a data field for each line of text, split at CR, LF or CRLF as a reader splits them.
(define (lines! [b (ptr buffer)] [text string]))
