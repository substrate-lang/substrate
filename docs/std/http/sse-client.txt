;; std/http/sse-client
;; Server-sent events read as the HTML standard's EventSource reads them, over the fetch module, so on a page or a
;; server alike. sse-client-open! makes a request and gives on-event each event of the answer, after on-open. on-closed
;; is called when the answer ends, with sse-ended; with sse-lost when the request or its body failed, its error in
;; error; or with sse-refused when the answer was not a 200 event stream, its status in status. Unless it was refused, a
;; client with reconnect set makes the request again after retry milliseconds, 3 seconds when 0 or as the server last
;; set it, with the last event's id, until sse-client-close!. method, url, headers and body must last until then.
;; requires host/events host/stream host/modules/timer host/modules/fetch std/http/http

(const sse-ended i32 0)

(const sse-lost i32 1)

(const sse-refused i32 2)

;; An event's strings last until on-event returns.
(struct sse-event [type string] [data string] [id string])

;; ---- the parser ----

;; A parser gives on-event each event of a stream fed to it in pieces, and sse-parser-end! readies it for the next
;; stream. id is the last event's id, kept across streams, and retry the last reconnection time a stream set, or -1.
(struct sse-parser [line buffer] [data buffer] [type buffer] [next-id buffer] [id buffer] [retry i64] [started bool]
  [after-cr bool] [stopped bool] [user (ptr void)] [on-event (fn ((ptr sse-parser) (ptr sse-event)))])

(define (sse-parser-init! [p (ptr sse-parser)] [user (ptr void)] [on-event (fn ((ptr sse-parser) (ptr sse-event)))]))

;; sse-parser-free! may be called from on-event, which stops sse-parse!.
(define (sse-parser-free! [p (ptr sse-parser)]))

;; An event whose blank line has not come is dropped.
(define (sse-parser-end! [p (ptr sse-parser)]))

;; Lines end at CR, LF or CRLF, a CRLF perhaps split between pieces.
(define (sse-parse! [p (ptr sse-parser)] [b (slice u8)]))

;; ---- the client ----

(struct sse-client [head event-head] [user (ptr void)] [reconnect bool] [retry f64] [status i32] [error i32]
  [method string] [url string] [headers (slice string)] [body (slice u8)] [state i32] [fetch u32] [timer u32]
  [parser sse-parser] [chunk (array u8 16384)]
  [on-open (fn ((ptr sse-client)))] [on-event (fn ((ptr sse-client) (ptr sse-event)))]
  [on-closed (fn ((ptr sse-client) i32))])

(define (sse-client-open! [c (ptr sse-client)] [method string] [url string] [headers (slice string)]
                          [body (slice u8)]))

;; sse-client-close! stops the client without calling on-closed.
(define (sse-client-close! [c (ptr sse-client)]))

;; ==== private ====

(const state-closed i32 0)

(const state-connecting i32 1)

(const state-open i32 2)

(const state-waiting i32 3)

(const default-retry f64 3000.0)

;; ---- the parser ----

(define (copy! [to (ptr buffer)] [s string]))

(define (dispatch! [p (ptr sse-parser)]))

(define (has-nul? [s string]) -> bool)

(define (field! [p (ptr sse-parser)] [name string] [value string]))

(define (line! [p (ptr sse-parser)]))

;; ---- the client ----

(define (parsed [p (ptr sse-parser)] [e (ptr sse-event)]))

(define (connect! [c (ptr sse-client)]))

(define (stop! [c (ptr sse-client)]))

(define (ended! [c (ptr sse-client)] [why i32]))

(define (answered! [c (ptr sse-client)] [status i32]))

(define (client-heard [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))
