;; std/http/multipart
;; multipart/form-data bodies (RFC 7578), after Go's mime/multipart. A part's file name has no directories, and its
;; bytes are views of the body.
;; requires std/http/http std/http/url

(struct multipart-part [name string] [filename string] [type string] [bytes (slice u8)] [file bool])

;; A multipart/form-data content type's boundary; "" for another type.
(define (multipart-boundary [a (ptr arena)] [content-type string]) -> string)

;; A body's named parts, in the arena; #f with the reason in why when the body is malformed.
(define (multipart-parse [a (ptr arena)] [body (slice u8)] [boundary string] [to (ptr (slice multipart-part))]
                         [why (ptr string)]) -> bool)

;; ==== private ====

(define (refused [why (ptr string)] [message string]) -> bool)

(define (at? [body (slice u8)] [i u32] [s string]) -> bool)

;; Where s next starts in body from i, or -1.
(define (find [body (slice u8)] [s string] [i u32]) -> i64)

(define (joined [a (ptr arena)] [front string] [back string]) -> string)

(define (base-name [s string]) -> string)

;; An RFC 5987 value, charset'language'escaped, decoded; "" unless its charset is UTF-8 or US-ASCII.
(define (extended [a (ptr arena)] [s string]) -> string)

(define (head! [a (ptr arena)] [text string] [part (ptr multipart-part)]))
