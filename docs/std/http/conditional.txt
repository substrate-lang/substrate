;; std/http/conditional
;; Conditional requests, RFC 9110's preconditions checked in its order, as Go's ServeContent checks them.
;; http-conditional! gives the answer the validators etag, unless empty, and modified, the time of the last change in
;; milliseconds since 1970, unless 0; then it answers 412 when If-Match or If-Unmodified-Since fails, or 304 to GET or
;; HEAD when If-None-Match or If-Modified-Since finds nothing new, and returns #t when it has answered. Headers added
;; to the answer before, such as its Cache-Control, go with a 304 too. http-if-range? tells whether a Range header
;; holds: when there is no If-Range, or it names the same strong ETag or the same date.
;; requires std/http/http std/http/server

(define (http-conditional! [r (ptr http-request)] [etag string] [modified f64]) -> bool)

(define (http-if-range? [r (ptr http-request)] [etag string] [modified f64]) -> bool)

;; ==== private ====

;; The entity tag at the start of s, W/"..." or "...", with the rest of s after it; "" when there is none.
(define (etag-at [s string] [rest (ptr string)]) -> string)

(define (weak? [tag string]) -> bool)

(define (opaque [tag string]) -> string)

;; matches? tells whether a list of entity tags, or *, names etag: by strong comparison, or else by weak.
(define (matches? [list string] [etag string] [strong bool]) -> bool)

;; The time a date header names, in milliseconds since 1970, or -1 when there is none or it is not a date.
(define (date-of [r (ptr http-request)] [name string]) -> f64)

;; Dates in headers count whole seconds.
(define (second-of [ms f64]) -> f64)
