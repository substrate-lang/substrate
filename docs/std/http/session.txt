;; std/http/session
;; Sessions in a signed cookie, after Flask's: the session's bytes, the second they were signed, and an HMAC-SHA256 of
;; both, each in base64url and joined by dots, as itsdangerous signs. A client can read its session but not change it.
;; The first of the keys signs, and any of them is accepted, so that keys can be rotated; each signs through a key
;; HKDF derives for sessions. The session form, (session T ...), gives each struct type T with a json form a reader
;; and a writer of a session holding it: T.session-read, which reads into the request's arena, and T.session-write!.
;; requires host/modules/clock std/crypto/hmac std/encoding/base64 std/encoding/json std/http/http std/http/server std/http/cookie compiler/macros std/meta/derive

;; keys are secrets, the newest first, and an empty one signs nothing and accepts nothing. name is the cookie's,
;; "session" when empty, and max-age how many seconds a session lasts, 31 days when 0. The cookie is HttpOnly,
;; SameSite=Lax, for the path /, and Secure over https.
(struct http-sessions [keys (slice string)] [name string] [max-age i64])

;; http-session-read gives the session's bytes, in the request's arena, when the cookie is signed by one of the keys
;; and has not expired.
(define (http-session-read [s (ptr http-sessions)] [r (ptr http-request)] [into (ptr (slice u8))]) -> bool)

;; http-session-write! sets the session's cookie to the bytes, signed now, before the request is answered. A session
;; too large for a cookie traps.
(define (http-session-write! [s (ptr http-sessions)] [r (ptr http-request)] [bytes (slice u8)]))

(define (http-session-clear! [s (ptr http-sessions)] [r (ptr http-request)]))

(define-syntax session)

;; ==== private ====

(define (name-of [s (ptr http-sessions)]) -> string)

(define (max-age-of [s (ptr http-sessions)]) -> i64)

(define (signed! [secret string] [text string] [mac (slice u8)]))

(define (now-seconds) -> u64)

(define (cookie! [s (ptr http-sessions)] [r (ptr http-request)] [value string] [max-age i64]))

(define (session-form [s datum]) -> datum)

(define (definitions [t datum] [info datum]) -> datum)
