;; std/http/http
;; HTTP/1.1 messages, parsed and written. The parser calls on-head, then on-body for each piece of the body, then
;; on-end, for each message it is fed. Header names are lowercased, and the head's names and values last until on-end.

;; A head's names and values have room for headers that grows as they come, up to max-headers.
(struct http-head [line buffer] [names (ptr string)] [values (ptr string)] [room u32] [count u32]
  [method string] [target string] [version string] [status i32] [reason string])

(struct http-parser [head http-head] [line buffer] [state i32] [remaining i64] [excess i64] [user (ptr void)]
  [on-head (fn ((ptr http-parser)))] [on-body (fn ((ptr http-parser) (slice u8)))] [on-end (fn ((ptr http-parser)))])

;; ---- text ----

(define (http-same? [a string] [b string]) -> bool)

;; http-origin-host? tells whether an Origin header's value, scheme://host, names the host in a Host header's value.
(define (http-origin-host? [origin string] [host string]) -> bool)

;; http-has-token? tells whether part is one of the comma-separated tokens in s, ignoring case.
(define (http-has-token? [s string] [part string]) -> bool)

;; http-header-has? tells whether any header of that name has part among its tokens.
(define (http-header-has? [h (ptr http-head)] [name string] [part string]) -> bool)

(define (http-trimmed [s string]) -> string)

;; A content type's media type, before its parameters.
(define (http-media-type [value string]) -> string)

(define (http-media-type? [value string] [type string]) -> bool)

;; A content type's parameter, ignoring case, its quotes and escapes removed into the arena; "" when it has none.
(define (http-media-parameter [a (ptr arena)] [value string] [name string]) -> string)

(define (http-header-count [h (ptr http-head)] [name string]) -> u32)

(define (http-decimal [s string]) -> i64)

;; Integers written in decimal, within least and most.
(define (http-unsigned [s string] [most u64] [to (ptr u64)]) -> bool)

(define (http-signed [s string] [least i64] [most i64] [to (ptr i64)]) -> bool)

(define (http-add-decimal! [b (ptr buffer)] [n i64]))

;; ---- the head ----

(define (http-header [h (ptr http-head)] [name string]) -> string)

;; to becomes a copy of from with its own text and room for just its headers, which http-head-free! frees.
(define (http-head-copy! [to (ptr http-head)] [from (ptr http-head)]))

(define (http-head-free! [h (ptr http-head)]))

;; ---- the parser ----

(define (http-parser-init! [p (ptr http-parser)] [user (ptr void)] [on-head (fn ((ptr http-parser)))]
                           [on-body (fn ((ptr http-parser) (slice u8)))] [on-end (fn ((ptr http-parser)))]))

(define (http-feed! [p (ptr http-parser)] [b (slice u8)]) -> i32)

(define (http-parser-closed! [p (ptr http-parser)]))

;; Called from on-head or on-end, http-parser-stop! makes http-feed! return how many bytes it took. After on-end,
;; http-parser-resume! starts on the next message.
(define (http-parser-stop! [p (ptr http-parser)]))

(define (http-parser-resume! [p (ptr http-parser)]))

;; Called from on-head, http-parser-refuse! makes http-feed! return -1, as for a malformed message.
(define (http-parser-refuse! [p (ptr http-parser)]))

;; Called from on-head, http-parser-no-body! ends the message at its head, as for a response to HEAD.
(define (http-parser-no-body! [p (ptr http-parser)]))

;; ---- writing heads ----

(define (http-reason [status i32]) -> string)

(define (http-write-request! [b (ptr buffer)] [method string] [target string]))

(define (http-write-status! [b (ptr buffer)] [status i32]))

(define (http-write-header! [b (ptr buffer)] [name string] [value string]))

(define (http-write-number! [b (ptr buffer)] [name string] [n i64]))

;; http-write-date! writes the date header for ms milliseconds since 1970.
(define (http-write-date! [b (ptr buffer)] [ms f64]))

;; ---- dates ----

;; http-add-date! adds the date ms milliseconds since 1970, as in Tue, 14 Nov 2023 22:13:20 GMT.
(define (http-add-date! [b (ptr buffer)] [ms f64]))

;; http-date reads a date in any of HTTP's three forms, as in Sun, 06 Nov 1994 08:49:37 GMT, Sunday, 06-Nov-94
;; 08:49:37 GMT and Sun Nov  6 08:49:37 1994, into ms milliseconds since 1970, and returns #f for anything else. A
;; two-digit year from 69 on is in the 1900s.
(define (http-date [s string] [ms (ptr f64)]) -> bool)

(define (http-write-end! [b (ptr buffer)]))

;; http-write-chunk! writes nothing for an empty chunk, and http-write-last-chunk! ends the body. http-write-chunk-size!
;; writes the line before a chunk's bytes, which a line end follows.
(define (http-write-chunk! [b (ptr buffer)] [bytes (slice u8)]))

(define (http-write-chunk-size! [b (ptr buffer)] [n u32]))

(define (http-write-last-chunk! [b (ptr buffer)]))

;; ==== private ====

(const max-head u32 16384)

(const max-line u32 1024)

(const max-excess i64 16384)

(const state-refused i32 8)

(const state-no-body i32 9)
