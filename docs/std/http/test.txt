;; std/http/test
;; Requests to a server through a connection without a socket, after Go's httptest. http-test! and http-test-raw!
;; return #t when the answer has come by the time they return. http-test-later! calls done once the answer comes, or
;; with status 0 once the server ends the connection without one; it is for handlers that answer later or trap. Each
;; request frees the one before, and http-test-free! the last.
;; requires host/events host/modules/timer std/http/http std/http/server

(struct http-test [head event-head] [connection (ptr void)] [parser http-parser] [request buffer] [answer http-head]
  [body buffer] [status i32] [bodiless bool] [over bool] [done (fn ((ptr http-test)))])

(define (http-test-status [t (ptr http-test)]) -> i32)

(define (http-test-header [t (ptr http-test)] [name string]) -> string)

(define (http-test-body [t (ptr http-test)]) -> (slice u8))

;; headers are names each followed by its value. A host header is added when there is none, and a content-length
;; unless the request is a GET or HEAD without a body.
(define (http-test! [t (ptr http-test)] [s (ptr http-server)] [method string] [target string] [headers (slice string)]
                    [body (slice u8)]) -> bool)

(define (http-test-later! [t (ptr http-test)] [s (ptr http-server)] [method string] [target string]
                          [headers (slice string)] [body (slice u8)] [done (fn ((ptr http-test)))]))

;; bodiless tells that the answer has no body, as for HEAD.
(define (http-test-raw! [t (ptr http-test)] [s (ptr http-server)] [bytes (slice u8)] [bodiless bool]) -> bool)

(define (http-test-free! [t (ptr http-test)]))

;; ==== private ====

(define (written! [t (ptr http-test)] [method string] [target string] [headers (slice string)] [body (slice u8)]))

(define (start! [t (ptr http-test)] [s (ptr http-server)] [bodiless bool]) -> bool)

(define (close! [t (ptr http-test)]))

(define (sink [tag (ptr void)] [b (slice u8)]))

(define (sink-ended [tag (ptr void)]))

;; A 100 Continue comes before the answer.
(define (answer-head [p (ptr http-parser)]))

(define (answer-body [p (ptr http-parser)] [b (slice u8)]))

(define (answer-end [p (ptr http-parser)]))

(define (over! [t (ptr http-test)] [status i32]))

(define (test-heard [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))
