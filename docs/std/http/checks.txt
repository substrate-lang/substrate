;; std/http/checks
;; Checks for a server's list, run before its handler: cross-origin protection, after Go's CrossOriginProtection;
;; security headers, Helmet's; CORS, after Express's cors; rate limits, a token bucket for each client's address,
;; after Go's x/time/rate; and allowed hosts, after Django's ALLOWED_HOSTS.
;; requires host/modules/clock std/http/http std/http/server

;; ---- cross-origin protection ----

;; trusted lists origins, as https://example.com, whose requests pass.
(struct http-cross-origin [trusted (slice string)])

;; http-cross-origin-check answers 403 to a browser's request from another origin, unless it is GET, HEAD or OPTIONS: by
;; Sec-Fetch-Site, or else by Origin against Host. A request with neither header passes.
(define (http-cross-origin-check [r (ptr http-request)] [settings (ptr void)]) -> bool)

;; ---- security headers ----

;; policy, when not empty, is the Content-Security-Policy, left to the program.
(struct http-security [policy string])

;; http-security-check adds Helmet's headers to every answer, Strict-Transport-Security over https alone.
(define (http-security-check [r (ptr http-request)] [settings (ptr void)]) -> bool)

;; ---- CORS ----

;; origins lists the origins allowed, or "*" for any, which cannot go with credentials. methods and headers are what a
;; preflight allows: when empty, Express's methods and the headers asked for. expose lists headers a page may read, and
;; max-age, when not 0, is how many seconds a preflight's answer may be kept.
(struct http-cors [origins (slice string)] [methods string] [headers string] [expose string] [credentials bool]
  [max-age u32])

;; http-cors-check answers a preflight with 204, and adds the CORS headers to every other answer, with Vary: Origin
;; unless any origin is allowed.
(define (http-cors-check [r (ptr http-request)] [settings (ptr void)]) -> bool)

;; ---- rate limits ----

;; rate is the requests a client may make each second and burst how many at once, both above 0; an IPv6 client is known
;; by its first 56 bits. most bounds the clients kept, 10000 at 0; a table three quarters full starts again.
(struct http-limit [rate f64] [burst u32] [most u32] [size u32] [count u32] [buckets (ptr bucket)])

;; http-limit-check answers 429, with Retry-After in seconds, a request whose client has no request left.
(define (http-limit-check [r (ptr http-request)] [settings (ptr void)]) -> bool)

;; ---- allowed hosts ----

;; names lists the hosts requests may name: a name, in any case; a name after a dot, for it and the names under it; or
;; "*" for any. addresses lets any IP address pass.
(struct http-hosts [names (slice string)] [addresses bool])

;; http-hosts-check answers 403 to a request whose Host is not allowed, against DNS rebinding; one without a Host passes.
(define (http-hosts-check [r (ptr http-request)] [settings (ptr void)]) -> bool)

;; ==== private ====

;; ---- cross-origin protection ----

(define (listed? [list (slice string)] [s string]) -> bool)

;; ---- CORS ----

(define (http-length-header! [r (ptr http-request)] [name string] [n i64]))

;; ---- rate limits ----

(struct bucket [key (array u8 46)] [length u32] [used bool] [tokens f64] [at f64])

;; The first 12 bytes of an IPv4 address mapped into IPv6.
(define mapped (array u8 12))

;; ipv4! reads a dotted IPv4 address into 4 bytes, or returns #f.
(define (ipv4! [s string] [out (ptr u8)]) -> bool)

;; ipv6! reads an IPv6 address (RFC 4291) into 16 bytes, ignoring a zone after %, or returns #f.
(define (ipv6! [s string] [out (ptr u8)]) -> bool)

;; key! writes the client's key and returns its length: an IPv4 address, also one mapped into IPv6, or a colon and the
;; first 56 bits of an IPv6 address.
(define (key! [address string] [key (ptr u8)]) -> u32)

;; slot gives the key's bucket in the table, or the empty one where it would go.
(define (slot [buckets (ptr bucket)] [size u32] [key (slice u8)]) -> (ptr bucket))

;; The table starts with 64 buckets and doubles, up to most, when three quarters full.
(define (grow! [l (ptr http-limit)] [size u32]))

(define (bucket-of [l (ptr http-limit)] [key (slice u8)]) -> (ptr bucket))

;; ---- allowed hosts ----

;; host-of gives the host a Host header's value names, without its port and a final dot, or "" when it is malformed.
(define (host-of [value string]) -> string)

(define (allowed? [entry string] [host string]) -> bool)
