;; std/http/form
;; Forms read into types. (form T ...) defines T.form-read for each struct type T, filling its fields by name from a
;; query's or urlencoded body's pairs and a multipart body's parts. As with json, a missing field is refused unless it
;; is a pointer; a missing bool is false, as an unchecked checkbox's, and a slice takes every value of its name. Files
;; are http-uploads, views of the body.
;; requires std/http/http std/http/url std/http/multipart compiler/macros std/meta/derive

(struct http-upload [filename string] [type string] [bytes (slice u8)])

(struct form-error [message string] [field string])

;; ---- the form form ----

(define-syntax form)

;; ==== private ====

(define (refuse! [err (ptr form-error)] [message string] [field string]) -> bool)

;; The kth value of the name, from the pairs and then the parts that are not files.
(define (text [pairs (slice url-pair)] [parts (slice multipart-part)] [name string] [k u32] [to (ptr string)]) -> bool)

(define (texts [pairs (slice url-pair)] [parts (slice multipart-part)] [name string]) -> u32)

(define (file [parts (slice multipart-part)] [name string] [k u32] [to (ptr http-upload)]) -> bool)

(define (files [parts (slice multipart-part)] [name string]) -> u32)

(define (float! [s string] [to (ptr f64)]) -> bool)

(define (truth! [s string] [to (ptr bool)]) -> bool)

;; ---- the form form ----

(define unsigned-limits datum)

(define signed-limits datum)

(define (literal [d datum]) -> datum)

(define (form-form [s datum]) -> datum)

(define (definitions [t datum] [info datum]) -> datum)

(define (upload? [type datum]) -> bool)

(define (value? [type datum]) -> bool)

(define (check! [type datum] [name datum]))

;; Code filling place p, the field of the name and type.
(define (field-code [name datum] [type datum] [p datum]) -> datum)

;; Code reading the text s into place p, of the type.
(define (value-code [type datum] [s datum] [p datum] [key datum]) -> datum)
