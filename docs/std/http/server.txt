;; std/http/server
;; An HTTP/1.1 server over net. A handler answers with bytes, a stream, pieces as they come, or nothing, or takes the
;; connection's stream. Settings: host, 127.0.0.1 when empty; secure for TLS; most connections, no limit at 0; timeouts
;; in ms, the default at 0 and none below 0: idle-timeout 5 s (130 s behind proxies), head-timeout 60 s,
;; request-timeout 300 s, send-timeout 60 s; body-limit for http-read-body!, 1 MiB at 0; compress, a gzip level for
;; answers of a type that compresses, of 1 KB or more, and for event streams that ask; proxies, how many proxies' hops
;; of X-Forwarded headers to believe; checks, run before handle. problem holds why listening failed.
;; A request lasts until it is answered and its body has ended; on-done is called as it ends and must not trap. A trap
;; in a request's code answers 500, or ends the connection once the answer has begun, and goes to on-trap, the log, or
;; stderr.
;; requires host/events host/stream host/node/net host/node/signal host/modules/timer host/modules/clock host/modules/random std/http/http std/log std/compress/deflate std/compress/gzip

;; on-body gets each piece of the body, and on-end sees failed set for a malformed one; on-gone is called once if the
;; client leaves before the answer is finished, and on-drained when http-write! may write again after returning #f.
;; The arena is freed with the request. A 2xx or 304 answer carries etag unless empty, modified unless 0, and ranges.
(struct http-request [connection (ptr void)] [server (ptr void)] [method string] [target string] [head (ptr http-head)]
  [arena arena] [user (ptr void)] [route (ptr void)] [peer string] [ended bool] [failed bool] [answered bool]
  [sent bool] [done bool] [gone bool] [chunked bool] [depth u32] [outer (ptr http-request)] [status i32] [length i64]
  [written i64] [began f64] [deadline f64] [on-body (fn ((ptr http-request) (slice u8)))]
  [on-end (fn ((ptr http-request)))]
  [on-gone (fn ((ptr http-request)))] [on-drained (fn ((ptr http-request)))] [body buffer] [limit u32]
  [taken (fn ((ptr http-request) (slice u8)))] [added buffer] [own-head http-head] [etag string] [modified f64]
  [ranges bool] [compressible bool] [events bool] [encoded bool] [gzip i32] [deflater (ptr deflater)] [crc u32]
  [size u32])

;; A check runs before the handler and returns #t when it has answered.
(struct http-check [run (fn ((ptr http-request) (ptr void)) -> bool)] [settings (ptr void)])

;; A hook's run is called with its tag as its server shuts down, and again, forced, when the time to finish is up.
(struct http-hook [run (fn ((ptr void) bool))] [tag (ptr void)] [server (ptr void)] [next (ptr http-hook)]
  [previous (ptr http-hook)])

(struct http-server [head event-head] [listener u32] [port u32] [host string] [secure bool] [most u32]
  [idle-timeout f64] [head-timeout f64] [request-timeout f64] [send-timeout f64] [body-limit u32] [compress i32]
  [proxies u32] [checks (slice http-check)] [problem string] [user (ptr void)] [connections (ptr http-connection)]
  [hooks (ptr http-hook)] [stopping bool] [stop-timer u32] [stopped (fn ((ptr http-server)))] [log (ptr logger)]
  [handle (fn ((ptr http-request)))] [listening (fn ((ptr http-server)))] [on-trap (fn ((ptr http-request) string))]
  [on-done (fn ((ptr http-request)))])

;; http-length! declares the body's length, written as the head ends.
(define (http-length! [r (ptr http-request)] [n i64]))

;; http-write! sends a piece of the body, after the head the first time. It returns #f when the request is gone or the
;; connection is full; on-drained is called once there is room.
(define (http-write! [r (ptr http-request)] [b (slice u8)]) -> bool)

;; http-finish! ends the answer, with no body when nothing was written. A body shorter than its declared length closes
;; the connection.
(define (http-finish! [r (ptr http-request)]))

;; http-send! sends the whole body with its length, compressed when it may be.
(define (http-send! [r (ptr http-request)] [b (slice u8)]))

(define (http-answer-text! [r (ptr http-request)] [status i32] [text string]))

;; ---- relays ----

;; http-send-stream! sends n bytes of the stream as the body, or all of it when n is -1. A stream that ends before n
;; bytes closes the connection.
(define (http-send-stream! [r (ptr http-request)] [from u32] [n i64]))

;; http-take! sends the head alone, and gives the connection's stream, untagged, to the taker once the handler has
;; returned. The connection counts toward most until the stream closes.
(define (http-take! [r (ptr http-request)] [take taker] [tag (ptr void)]))

;; ---- reading bodies ----

;; http-read-body! gives taken the whole body once it has come. A body longer than limit, the server's when 0, is
;; answered with 413 and its connection closed; a malformed one with 400.
(define (http-read-body! [r (ptr http-request)] [limit u32] [taken (fn ((ptr http-request) (slice u8)))]))

;; ---- the server ----

(define (http-serve! [s (ptr http-server)] [port u32] [handle (fn ((ptr http-request)))]) -> u32)

;; http-server-close! stops listening and drops every connection at once, and runs the hooks forced.
(define (http-server-close! [s (ptr http-server)]))

;; ---- shutting down ----

;; http-hook! has the hook run when the server shuts down; http-unhook! takes it back as its owner ends.
(define (http-hook! [s (ptr http-server)] [h (ptr http-hook)]))

(define (http-unhook! [h (ptr http-hook)]))

;; http-server-shutdown! stops listening, drops idle connections, has the others close once their answers are done, and
;; runs the hooks; after ms milliseconds it drops what remains and runs the hooks forced. done is called once nothing
;; remains.
(define (http-server-shutdown! [s (ptr http-server)] [ms f64] [done (fn ((ptr http-server)))]))

;; ---- signals ----

;; http-stop-on-signals! shuts the server down at SIGTERM or SIGINT, with ms milliseconds to finish, then calls done,
;; or exits when done is 0; a second signal exits at once.
(define (http-stop-on-signals! [s (ptr http-server)] [ms f64] [done (fn ((ptr http-server)))]))

;; A relay sends a stream as an answer's body.

;; A connection from http-connect! has no stream: its bytes go to sink, and http-connection-close! frees it. started is

  [request (ptr http-request)] [block (ptr u8)] [out buffer] [relay (ptr relay)] [reading bool] [busy bool]
  [closing bool] [half-closed bool] [feeding bool] [full bool] [dropped bool] [take taker] [taken (ptr void)]
  [timer u32] [timer-at f64] [due f64] [rest (slice u8)] [next (ptr http-connection)] [previous (ptr http-connection)]
  [peer string] [started f64] [discarded u64] [lingering bool] [linger-end f64] [doomed bool]
  [sink (fn ((ptr void) (slice u8)))] [sink-ended (fn ((ptr void)))] [sink-tag (ptr void)])
