;; std/http/websocket
;; WebSocket servers (RFC 6455). ws-accept! upgrades a request, and the peer's messages and the end arrive as calls.
;; A peer silent for ping milliseconds, 25 s when 0 and never below 0, is pinged, and the connection ends if it stays
;; silent as long again. While what is written to a peer cannot go out, nothing more is read from it.
;; requires host/events host/stream host/modules/timer host/modules/clock std/http/http std/http/server std/crypto/sha1 std/encoding/base64 std/encoding/utf8

(const ws-text i32 1)

(const ws-binary i32 2)

(struct ws-connection [head event-head] [stream u32] [user (ptr void)] [server (ptr http-server)] [ping f64]
  [hook http-hook] [frames buffer] [message buffer] [out buffer]
  [kind i32] [state i32] [code i32] [reading bool] [full bool] [timer u32] [timer-at f64] [heard f64] [pinged bool]
  [any-origin bool]
  [on-message (fn ((ptr ws-connection) (slice u8) i32))]
  [on-closed (fn ((ptr ws-connection) i32))]
  [on-drained (fn ((ptr ws-connection)))])

(define (ws-upgrade? [r (ptr http-request)]) -> bool)

;; ---- the connection ----

;; ws-accept! returns #t when it accepts the request, after which the caller sets the callbacks and user. It returns #f
;; when it refuses the request: with 400 for a bad handshake, 426 for a version other than 13, and 403 for a page of
;; another origin unless any-origin is set. Given a protocol, it accepts with that subprotocol, and refuses with 400 a
;; request that does not offer it.
(define (ws-accept! [w (ptr ws-connection)] [r (ptr http-request)] [protocol string]) -> bool)

;; ws-send! returns what stream-write does, or -1 when the connection is not open.
(define (ws-send! [w (ptr ws-connection)] [b (slice u8)] [kind i32]) -> i32)

(define (ws-send-text! [w (ptr ws-connection)] [s string]) -> i32)

(define (ws-close! [w (ptr ws-connection)] [code i32]))

;; ==== private ====

;; A message beyond this many bytes closes the connection with 1009.
(const largest u32 16777216)

(const state-open i32 0)

(const state-closing i32 1)

(const state-finishing i32 2)

(const state-closed i32 3)

(define guid string)

(define (key? [k string]) -> bool)

(define (same-origin? [r (ptr http-request)]) -> bool)

(define (closing-code? [code i32]) -> bool)

;; ---- frames out ----

(define (write-frame! [w (ptr ws-connection)] [opcode i32] [payload (slice u8)]) -> i32)

(define (write-close! [w (ptr ws-connection)] [code i32]))

(define (disarm! [w (ptr ws-connection)]))

(define (ended! [w (ptr ws-connection)] [code i32]))

(define (finish! [w (ptr ws-connection)] [code i32]))

;; fail! sends a close and finishes the stream without waiting for the peer's close.
(define (fail! [w (ptr ws-connection)] [code i32]))

;; ---- frames in ----

(define (deliver! [w (ptr ws-connection)] [message (slice u8)] [kind i32]))

(define (frame! [w (ptr ws-connection)] [fin bool] [opcode i32] [payload (slice u8)]))

;; next-frame! handles the frame at p and returns its size, or returns 0 when the frame is not complete.
(define (next-frame! [w (ptr ws-connection)] [frames (slice u8)]) -> u32)

;; frames! handles the whole frames at the front of the bytes, and returns how many bytes they took.
(define (frames! [w (ptr ws-connection)] [b (slice u8)]) -> u32)

(define (unframe! [w (ptr ws-connection)]))

(define (heard! [w (ptr ws-connection)] [b (slice u8)]))

(define (interval [w (ptr ws-connection)]) -> f64)

(define (arm! [w (ptr ws-connection)] [ms f64]))

(define (read-more! [w (ptr ws-connection)]))

(define (lost! [w (ptr ws-connection)]))

;; silent!: a peer heard from within the interval is waited on for the rest of it; one silent for it is pinged, or ended
;; if it was pinged already. now is no earlier than the timer was set for.
(define (silent! [w (ptr ws-connection)]))

(define (ws-heard [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))

;; stopping closes the connection with 1001 as its server shuts down, or ends it when forced.
(define (stopping [tag (ptr void)] [forced bool]))

(define (taken [tag (ptr void)] [stream u32] [b (slice u8)]))

;; ---- the connection ----

(define (refuse! [r (ptr http-request)] [status i32]))

;; offered? tells whether the request offers protocol among its subprotocols, which are compared exactly.
(define (offered? [r (ptr http-request)] [protocol string]) -> bool)
