;; std/http/cookie
;; Cookies (RFC 6265, with SameSite and the name prefixes of its revision), after Go's net/http. http-request-cookie
;; gives a cookie's value from the request, skipping malformed pairs, and http-set-cookie! sets one.
;; requires std/http/http std/http/server

(const http-same-site-lax i32 1)

(const http-same-site-strict i32 2)

(const http-same-site-none i32 3)

;; max-age is in seconds: 0 leaves it out, and below 0 ends the cookie now. same-site 0 leaves it out.
(struct http-cookie [name string] [value string] [path string] [domain string] [max-age i64] [secure bool]
  [http-only bool] [same-site i32])

;; http-request-cookie gives the value of the first cookie of that name, or "".
(define (http-request-cookie [r (ptr http-request)] [name string]) -> string)

;; http-set-cookie! traps on a cookie that browsers would refuse: a bad name, value, path or domain; a name and value
;; longer than 4096 bytes; SameSite=None without Secure; a name starting __Secure-, in any case, without Secure; or one
;; starting __Host- without Secure, without the path / or with a domain.
(define (http-set-cookie! [r (ptr http-request)] [c (ptr http-cookie)]))

;; ==== private ====

(define separators string)

(define (name-byte? [c u8]) -> bool)

(define (name? [s string]) -> bool)

;; Reading allows a space or comma in a value; writing does not.
(define (value? [s string] [reading bool]) -> bool)

(define (unquoted [s string]) -> string)

(define (path? [s string]) -> bool)

(define (domain? [s string]) -> bool)

(define (prefixed? [s string] [prefix string]) -> bool)
