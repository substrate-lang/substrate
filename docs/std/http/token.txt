;; std/http/token
;; API tokens: random, shown to their owner once, kept as their SHA-256, and presented in an Authorization: Bearer
;; header (RFC 6750).
;; requires host/modules/random std/crypto/sha256 std/encoding/base64 std/http/http std/http/server

;; http-token-new! adds a new token to b: the prefix, then 32 random bytes in base64url.
(define (http-token-new! [b (ptr buffer)] [prefix string]))

;; http-token-hash! gives the token's SHA-256, the 32 bytes to keep.
(define (http-token-hash! [token string] [into (slice u8)]))

;; http-request-bearer gives the token of the request's Authorization header, or "".
(define (http-request-bearer [r (ptr http-request)]) -> string)
