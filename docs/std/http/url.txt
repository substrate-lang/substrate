;; std/http/url
;; The parts of a request's target: its path and query, percent-decoding and encoding, a path's segments, and the
;; application/x-www-form-urlencoded format of queries and form bodies, after the WHATWG URL Standard.

(struct url-pair [name string] [value string])

(define (url-path [target string]) -> string)

(define (url-query [target string]) -> string)

;; s with its percent-escapes decoded, and its pluses as spaces when plus is set, into the arena; s when it has none. A
;; % without two hex digits after it stays.
(define (url-unescape [a (ptr arena)] [s string] [plus bool]) -> string)

;; A path's segments after its first slash, decoded into the arena: "/" has none, and a final slash gives an empty last
;; segment. It returns 1 when a segment is . or .., or empty before the end, and -1 when the path does not start with a
;; slash.
(define (url-segments [a (ptr arena)] [path string] [to (ptr (slice string))]) -> i32)

;; The path the segments name, resolved as Go's path.Clean does, keeping a final slash.
(define (url-add-clean! [b (ptr buffer)] [segments (slice string)]))

;; s as one path segment, escaped as Go's PathEscape does.
(define (url-add-segment! [b (ptr buffer)] [s string]))

;; s as a name or value of the urlencoded format.
(define (url-add-form-value! [b (ptr buffer)] [s string]))

;; An urlencoded string's pairs, decoded into the arena.
(define (url-form-parse [a (ptr arena)] [s string]) -> (slice url-pair))

;; The value of the first pair with the name.
(define (url-form-value [pairs (slice url-pair)] [name string] [to (ptr string)]) -> bool)

;; ==== private ====

(define (escape? [s string] [i u32]) -> bool)

(define hex-digits string)

(define (add-escape! [b (ptr buffer)] [c u8]))

(define (alphanumeric? [c u8]) -> bool)
