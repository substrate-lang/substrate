;; std/http/static
;; Files: http-send-file! answers with a file under the root, as Flask's send_from_directory does, and http-static! with
;; the file at the request's path, a directory's index.html, or a redirect adding a final slash. Answers carry
;; Last-Modified and an ETag, are 304 or 412 as preconditions require, hold one range when asked, and have Cache-Control
;; no-cache unless cache gives another value for the path. A part that is . or .., or that starts with a dot other than
;; .well-known, is not found. An empty root is the current directory.
;; requires host/fs std/http/http std/http/server std/http/conditional std/http/url

(struct http-files [root string] [cache (fn (string) -> string)])

;; ---- answers ----

;; http-send-file! sends the file at path, relative to the root, its parts separated by slashes.
(define (http-send-file! [r (ptr http-request)] [files (ptr http-files)] [path string]))

(define (http-static! [r (ptr http-request)] [files (ptr http-files)]))

;; ==== private ====

;; Files up to this size are read and sent whole; larger ones are piped.
(const whole i64 262144)

;; The most ranges a Range header may list, and the most parts a path may have.
(const most-ranges u32 16)

(const most-parts u32 64)

(define (extension [path string]) -> string)

(define (content-type [path string]) -> string)

;; ---- paths ----

(define (hidden? [part string]) -> bool)

;; refusal gives the status for a path's parts: 400 when one holds a slash or NUL, 404 when one is hidden, else 0.
(define (refusal [parts (slice string)]) -> i32)

;; rooted! adds the root to b and returns its length.
(define (rooted! [b (ptr buffer)] [files (ptr http-files)]) -> u32)

(define (joined! [b (ptr buffer)] [parts (slice string)]))

(define (moved! [r (ptr http-request)]))

;; ---- ranges ----

(struct span [start i64] [end i64])

(define (span! [s (ptr span)] [start i64] [end i64]))

;; The number s is, or -1 when it is not all digits.
(define (position [s string]) -> i64)

;; gapless? tells whether the spans cover every byte from low up to high.
(define (gapless? [spans (ptr span)] [n u32] [low i64] [high i64]) -> bool)

;; ranges! reads a GET's Range header for size bytes into one span: 0 for the whole file, as with no header, one that
;; does not parse, or ranges that do not join into one; 1 for a span; -1 when no range falls within the file.
(define (ranges! [r (ptr http-request)] [size i64] [etag string] [modified f64] [to (ptr span)]) -> i32)

(define (content-range! [r (ptr http-request)] [s (ptr span)] [size i64]))

;; ---- answers ----

(define (etag! [b (ptr buffer)] [info (ptr fs-info)]))

(define (cache-of [files (ptr http-files)] [path string]) -> string)

(define (head-of! [r (ptr http-request)] [name string] [partial bool] [s (ptr span)] [size i64]))

(define (failed! [r (ptr http-request)]))

(define (body! [r (ptr http-request)] [name string] [partial bool] [s (ptr span)] [size i64]))

;; send! answers with the file at name, whose path under the root is path.
(define (send! [r (ptr http-request)] [files (ptr http-files)] [name string] [path string] [info (ptr fs-info)]))

(define (not-found! [r (ptr http-request)]))

;; read? tells whether the request may read files, and answers 405 when it may not.
(define (read? [r (ptr http-request)]) -> bool)
