;; std/doc/root
;; The documentation of a root, a package or a root of the toolchain, from its label, its modules' paths there and the
;; paths of those that are public: each module's documentation, with the package it is private to if it is, and each
;; directory's index: its name, its main module's header (the root's is label.sub), its entries, each a public module
;; in it or a directory in it with the first sentence of a header (a directory's from its main module), and its private
;; modules' entries.
;; requires compiler/macros std/doc/doc std/package/path

(define (root-doc [label datum] [modules datum] [public datum]) -> datum)

;; directory-index gives a directory's index from its modules' headers alone.
(define (directory-index [label datum] [dir string] [modules datum] [public datum]) -> datum)

(define (owner-of [label datum] [public datum] [rel datum]) -> datum)

;; ==== private ====

(define (identity-in [label datum] [rel datum]) -> datum)

(define (index-of [label datum] [dir string] [heads datum]) -> datum)

(define (main-of [label datum] [dir string]) -> datum)

(define (header-in [rel datum] [heads datum]) -> datum)

(define (directories [rels datum]) -> datum)

(define (child-of [dir string] [rel string]) -> string)

(define (parent-in [rel string]) -> string)
