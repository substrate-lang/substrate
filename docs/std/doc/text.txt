;; std/doc/text
;; Documentation as text: a module as its declarations would read in a header file, and a directory as a list of its
;; modules. A module's text is its name and header comment, its language, what it requires and re-exports, then its
;; public items and below them its private ones, each part under the module's section labels.
;; requires compiler/macros std/doc/doc

(define (module-text [d datum]) -> datum)

(define (item-text [i datum]) -> datum)

(define (index-text [name datum] [header datum] [entries datum]) -> datum)

;; ==== private ====

(define (part! [b (ptr buffer)] [items datum]))

(define (item! [b (ptr buffer)] [i datum]))

(define (commented! [b (ptr buffer)] [text datum]))

(define (listed! [b (ptr buffer)] [what string] [l datum]))
