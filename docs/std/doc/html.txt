;; std/doc/html
;; Documentation as HTML pages, one per module and one per directory, at each one's name under the output directory: a
;; module's page has its items under the module's sections, public then private, with a sidebar of them by kind; a
;; directory's lists its modules and directories. Pages link to each other, and a type's name in a signature links to
;; where the type is defined.
;; requires compiler/macros std/doc/doc std/web/html std/package/path

;; module-page takes the names of the documented modules, and for each type a signature may name, the module defining
;; it.
(define (module-page [d datum] [names datum] [types datum]) -> datum)

(define (index-page [index datum]) -> datum)

(define (top-page [labels datum]) -> datum)

;; ---- paths ----

(define page-style string)

;; ==== private ====

;; ---- the page ----

(define (opened! [b (ptr buffer)] [title string] [dir string] [path string]))

(define (closed! [b (ptr buffer)]))

(define (paragraphs! [b (ptr buffer)] [text datum]))

(define (meta! [b (ptr buffer)] [d datum] [dir string] [names datum]))

(define (named! [b (ptr buffer)] [name datum] [dir string] [names datum]))

;; page-of gives the documented module a name stands for, as a name for a directory stands for its main module.
(define (page-of [name datum] [names datum]) -> datum)

;; ---- items ----

(define (sidebar! [b (ptr buffer)] [items datum]))

(define (part! [b (ptr buffer)] [items datum] [dir string] [module string] [types datum]))

(define (item! [b (ptr buffer)] [i datum] [dir string] [module string] [types datum]))

(define (signature! [b (ptr buffer)] [t string] [dir string] [module string] [types datum] [own string]))

(define (delimiter? [c u8]) -> bool)

;; ---- paths ----

(define (directory-of [name string]) -> string)

;; link gives the path from the directory dir to the file to, both under the output directory, which "" is.
(define (link [dir string] [to string]) -> string)

(define (slash-in [p string]) -> u32)

(define (after-slash [p string]) -> string)

(define (joined [x string] [y string]) -> string)
