;; std/doc/html
;; Documentation as HTML pages, one per module and one per directory, at each one's name under the output directory: a
;; module's page has its items under the module's sections, public then private, with a sidebar of them by kind; a
;; directory's lists its modules and directories, the private modules below. Pages link to each other, and a type's
;; name in a signature links to where the type is defined. Each page is a tree std/html writes.
;; requires compiler/macros std/doc/doc std/html/html std/package/path

;; html-files gives the files of roots' documentation, each path with its page: index.html listing the roots, each module's
;; and each directory's page at its name, and style.css.
(define (html-files [roots datum]) -> datum)

;; ==== private ====

;; types-for gives each type a signature in a module may name, its own and those of the modules it requires, with the
;; module that defines it.
(define (types-for [d datum] [docs datum]) -> datum)

;; ---- pages ----

;; module-page takes the names of the documented modules, for each type a signature may name the module defining it,
;; and the package the module is private to, if it is.
(define (module-page [d datum] [names datum] [types datum] [owner datum]) -> datum)

(define (index-page [index datum]) -> datum)

(define (top-page [labels datum]) -> datum)

(define (page [title string] [dir string] [path string] [body datum]) -> datum)

(define (crumbs [dir string] [path string]) -> datum)

(define (paragraphs [text datum]) -> datum)

(define (meta [d datum] [dir string] [names datum] [owner datum]) -> datum)

(define (entries [name string] [es datum]) -> datum)

(define (entry-page [n string]) -> string)

(define (row [first datum] [second datum]) -> datum)

(define (privately [items datum] [trees datum]) -> datum)

(define (named [name datum] [dir string] [names datum]) -> datum)

(define (linked [href datum] [child datum]) -> datum)

(define (code [x datum]) -> datum)

(define (separated [l datum]) -> datum)

;; ---- items ----

(define (sidebar [items datum]) -> datum)

(define (part [items datum] [dir string] [module string] [types datum]) -> datum)

(define (item [i datum] [dir string] [module string] [types datum]) -> datum)

;; signature gives a form's text as children of a tree, each type's name in it linked to where the type is defined.
(define (signature [t string] [dir string] [module string] [types datum] [own string]) -> datum)

(define (delimiter? [c u8]) -> bool)

;; ---- paths ----

(define (directory-of [name string]) -> string)

;; link gives the path from the directory dir to the file to, both under the output directory, which "" is.
(define (link [dir string] [to string]) -> string)

(define (slash-in [p string]) -> u32)

(define (after-slash [p string]) -> string)

(define (joined [x string] [y string]) -> string)

(define page-style string)
