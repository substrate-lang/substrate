;; std/meta/type-info
;; Types as data at compile time, for macros that write code from a type's fields.
;; requires compiler/macros

;; (kind name items parameters), kind struct, union or variant, or 0 when t names none. A struct's or union's items are
;; (field . type), a variant's (case (field . type) ...). A type is i8 to u64, f32, f64, bool or string, (ptr type),
;; (slice type), (array type n), a struct's, union's or variant's identifier, or (other type).
(define (type-info [t datum]) -> datum)

;; ==== private ====

(define (binding-of [t datum]) -> datum)

(define (case-info [c datum]) -> datum)

(define (field-info [f datum]) -> datum)

(define scalars datum)

(define (type-of [t datum]) -> datum)

;; A type as the checker holds it.
(define (parsed-type [d datum]) -> datum)
