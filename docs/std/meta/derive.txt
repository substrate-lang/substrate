;; std/meta/derive
;; Deriving definitions from types: a form's types and the types they reach, each given its definitions once.
;; requires compiler/macros std/meta/type-info

;; derive expands s, (who type ...), to the definitions code gives each type and the types they reach.
(define (derive [s datum] [who datum] [marker-name datum] [code (fn (datum datum) -> datum)]) -> datum)

;; t's definition named what, as type.what in the form's scope.
(define (derive-name [t datum] [what datum]) -> datum)

(define (derive-want! [t datum]))

(define (derive-fresh [base datum]) -> datum)

;; A type-info type as the type it is.
(define (type-syntax [type datum]) -> datum)

;; ==== private ====

(define context datum)

(define marker datum)

(define wanted datum)

(define counter u32)
