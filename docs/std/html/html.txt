;; std/html/html
;; HTML: the elements and attributes of the standard, and writing it: an element's start tag is html-open!, its
;; attributes, then html-opened!, and its end tag html-close!, or a whole tree is html-tree!, written as ui writes a view:
;; text, or (element ((name value) (name) ...) child ...), the attributes optional. Text and attribute values are
;; escaped.
;; requires compiler/macros

;; ---- the vocabulary ----

(define (html-element? [name string]) -> bool)

(define (void-element? [name string]) -> bool)

(define (custom-element? [name string]) -> bool)

(define (attribute-name? [name string]) -> bool)

(define (attribute-of? [element string] [name string]) -> bool)

(define (boolean-attribute? [name string]) -> bool)

;; ---- writing ----

(define (html-open! [b (ptr buffer)] [element string]))

(define (html-attribute! [b (ptr buffer)] [name string] [value string]))

(define (html-flag! [b (ptr buffer)] [name string]))

(define (html-opened! [b (ptr buffer)]))

(define (html-close! [b (ptr buffer)] [element string]))

(define (html-text! [b (ptr buffer)] [s string]))

(define (html-comment! [b (ptr buffer)] [s string]))

(define (html-tree! [b (ptr buffer)] [tree datum]))

;; ==== private ====

;; ---- the vocabulary ----

(define elements datum)

(define void-elements datum)

(define global-attributes datum)

(define element-attributes datum)

(define boolean-attributes datum)

(define (prefixed? [s string] [prefix string]) -> bool)

;; ---- writing ----

(define (escaped! [b (ptr buffer)] [s string] [attribute bool]))
