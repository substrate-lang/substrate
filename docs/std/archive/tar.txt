;; std/archive/tar
;; Tar archives, read from whole buffers: POSIX ustar, with pax extended headers for long paths and sizes, and GNU's long
;; names. tar-next! gives each entry in turn: a file with its bytes, a directory, or another kind, with its path.
;; requires host/fs

(const tar-file i32 1)

(const tar-directory i32 2)

(const tar-other i32 3)

(struct tar-entry [kind i32] [path string] [bytes (slice u8)])

(struct tar-reader [data (slice u8)] [at u32])

(define (tar-reader-init! [r (ptr tar-reader)] [data (slice u8)]))

;; An entry's path is in a or the data, and a file's bytes are in the data.
(define (tar-next! [a (ptr arena)] [r (ptr tar-reader)] [e (ptr tar-entry)] [err (ptr buffer)]) -> i32)

;; tar-unpack! gives 0, or -1 when data is not a tar archive and -2 when a file cannot be written, with err saying why.
(define (tar-unpack! [a (ptr arena)] [data (slice u8)] [dir string] [err (ptr buffer)]) -> i32)
