;; std/web/examples/snake/rules
;; The game snake, without its page: the rules, the timing and the placing of food.

;; A snake is at most 400 cells long, the size of the body array.
(const snake-columns i32 20)

(const snake-rows i32 20)

(struct cell [x i32] [y i32])

(const snake-waiting u32 0)

(const snake-playing u32 1)

(const snake-paused u32 2)

(const snake-over u32 3)

(const snake-won u32 4)

;; Directions, in clockwise order.
(const snake-up u32 0)

(const snake-right u32 1)

(const snake-down u32 2)

(const snake-left u32 3)

(struct snake
  [body (array cell 400)] [head u32] [length u32]
  [direction u32]
  [turns (array u32 2)] [turn-count u32]
  [food cell]
  [score u32]
  [state u32]
  [random u32]
  [timed bool] [last-time f64] [pending f64])

(define (snake-segment [s (ptr snake)] [i u32]) -> (ptr cell))

(define (snake-occupied? [s (ptr snake)] [x i32] [y i32]) -> bool)

(define (snake-reset! [s (ptr snake)] [seed u32]))

(define (snake-turn! [s (ptr snake)] [direction u32]))

(define (snake-toggle-pause! [s (ptr snake)]))

(define (snake-step! [s (ptr snake)]))

(define (snake-advance! [s (ptr snake)] [now f64]) -> bool)

;; ==== private ====

(define (dx [direction u32]) -> i32)

(define (dy [direction u32]) -> i32)

(define (opposite [direction u32]) -> u32)

(define (covers? [s (ptr snake)] [x i32] [y i32] [n u32]) -> bool)

(define (next-random! [s (ptr snake)]) -> u32)

(define (place-food! [s (ptr snake)]))

(define (reverse! [s (ptr snake)]))

(define (step-length [score u32]) -> f64)

(const max-steps u32 5)
