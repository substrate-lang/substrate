;; std/web/examples/platformer/world
;; A small platformer: the game itself, without its page.
;; requires host/fs std/web/game/format std/web/game/trap std/web/game/store std/web/game/draw std/web/game/tiles std/web/game/input std/web/game/sound std/web/game/timers

;; ---- sizes ----

(define (screen-width) -> f64)

(define (screen-height) -> f64)

;; ---- the levels ----

(const tile-spikes u32 4)

;; ---- the world ----

(struct coin [x f64] [y f64])

(struct enemy [body body] [home f64] [direction f64] [flat bool])

(const world-playing u32 0)

(const world-dying u32 1)

(const world-cleared u32 2)

(const world-won u32 3)

(struct world
  [map tilemap]
  [camera camera]
  [sheet u32]
  [state u32]
  [steps u32] [deaths u32]
  [levels string] [level u32] [level-text string] [grid text-grid]
  [player body] [facing i32] [coyote u32] [jump-wanted u32]
  [coins (store coin)] [coin-count u32] [collected u32]
  [enemies (store enemy)]
  [timers (timers event)]
  [flag-x f64] [flag-y f64])

;; ---- loading a level ----

(define (world-start! [w (ptr world)] [directory string]))

(define (world-init! [w (ptr world)]))

;; ---- one step ----

(define (world-step! [w (ptr world)] [in (ptr input)]))

;; ---- drawing ----

(define (world-draw [w (ptr world)]))

;; ==== private ====

;; ---- the levels ----

;; Tile n is drawn as cell n - 1 of the sprite sheet.
(const tile-grass u32 1)

(const tile-dirt u32 2)

(const tile-brick u32 3)

;; Cells of assets/sprites.png, in the order art/art.txt lists them.
(const sprite-coin u32 4)

(const sprite-flag u32 7)

(const sprite-stand u32 8)

(const sprite-run u32 9)

(const sprite-jump u32 11)

(const sprite-enemy u32 12)

(const sprite-enemy-flat u32 14)

(const sprite-cloud u32 15)

;; ---- how things move ----

(define (gravity) -> f64)

(define (fall-speed) -> f64)

(define (run-speed) -> f64)

(define (run-push) -> f64)

(define (run-drag) -> f64)

(define (jump-speed) -> f64)

(define (jump-cut) -> f64)

(define (bounce-speed) -> f64)

(define (enemy-speed) -> f64)

;; ---- the world ----

(variant event
  [enemy-gone [who (handle enemy)]]
  [restart]
  [next-level])

;; ---- loading a level ----

(define (follow-player! [w (ptr world)]))

(define (read-level! [w (ptr world)] [n u32] [err (ptr string)]) -> i32)

(define (load-level! [w (ptr world)]))

;; ---- one step ----

(define (smaller [a f64] [b f64]) -> f64)

(define (larger [a f64] [b f64]) -> f64)

(define (die! [w (ptr world)]))

(define (step-player! [w (ptr world)] [in (ptr input)]))

(define (step-enemy! [w (ptr world)] [e (ptr enemy)]))

(define (touch! [w (ptr world)]))

(define (happen! [w (ptr world)] [e (ptr event)]))

;; ---- drawing ----

(define (sprite [w (ptr world)] [frame u32] [x f64] [y f64] [flip bool]))
