;; requires std/web/gpu compiler/macros std/web/shade/core

;; ---- run-time support ----

  [depth-format u32] [depth-compare u32] [depth-write bool]
  [target-count u32] [targets (array u32 4)]
  [groups u32])

(define (render-settings-init! [s (ptr render-settings)]))
