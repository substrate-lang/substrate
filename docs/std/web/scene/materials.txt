;; std/web/scene/materials
;; Materials painted by the GPU from noise, into one texture of many layers with mip levels.
;; A material is a greyscale, which a surface's own colour multiplies.
;; requires std/web/gpu std/web/scene/shader std/web/shade/screen std/web/shade/noise

(const material-grass u32 1)

(const material-dirt u32 2)

(const material-rock u32 3)

(const material-sand u32 4)

(const material-snow u32 5)

(const material-bark u32 6)

(const material-leaves u32 7)

(const material-blades u32 8)

(const material-planks u32 9)

(const material-tiles u32 10)

(const material-plaster u32 11)

(const material-needles u32 12)

(const material-face u32 13)

(const material-count u32 14)

(struct materials [texture u32] [sampler u32] [shader paint-shader] [brushes u32] [group u32] [pipeline u32])

(define (material-scale [m u32]) -> f64)

(define (materials-init! [m (ptr materials)]))

(define (materials-texture [m (ptr materials)]) -> u32)

(define (materials-sampler [m (ptr materials)]) -> u32)
