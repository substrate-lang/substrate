;; std/web/scene/sprites
;; Sprites drawn by the GPU: a frame's sprites gathered into a batch and drawn in a few commands.
;; requires std/web/gpu std/web/scene/shader

(struct sprites
  [items (ptr f32)] [count u32] [capacity u32]
  [runs (ptr run)] [run-count u32] [run-capacity u32]
  [sheets (ptr sheet)] [sheet-count u32] [sheet-capacity u32]
  [shader quad-shader] [sampler u32]
  [uniforms u32] [screen-group u32]
  [buffer u32] [buffer-capacity u32]
  [pipelines (array u32 10)])

(define (sprites-init! [b (ptr sprites)]))

(define (sprites-sheet! [b (ptr sprites)] [path string]) -> u32)

(define (sprites-begin! [b (ptr sprites)]))

(define (sprites-add! [b (ptr sprites)] [sheet u32] [frame u32] [cell-width f64] [cell-height f64] [flip bool]
                      [x f64] [y f64] [w f64] [h f64] [color u32]))

(define (sprites-rect! [b (ptr sprites)] [x f64] [y f64] [w f64] [h f64] [color u32]))

(define (sprites-draw! [b (ptr sprites)] [kind u32] [target u32] [format u32] [width u32] [height u32] [load u32] [clear u32]))

;; ==== private ====

(struct run [sheet u32] [first u32] [count u32])

(struct sheet [texture u32] [group u32])

(const floats-per-instance u32 12)

(const bytes-per-instance u32 48)

(define (add-sheet! [b (ptr sprites)] [texture u32]) -> u32)

(define (join-run! [b (ptr sprites)] [sheet u32]))

(define (next-instance! [b (ptr sprites)]) -> (ptr f32))

(define (instance! [b (ptr sprites)] [sheet u32] [frame u32] [cell-width f64] [cell-height f64] [flip bool]
                   [x f64] [y f64] [w f64] [h f64] [color u32]))

(define (pipeline! [b (ptr sprites)] [format u32]) -> u32)

(define (part [color u32] [shift u32]) -> f64)
