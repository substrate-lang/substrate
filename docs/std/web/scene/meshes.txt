;; std/web/scene/meshes
;; Meshes drawn by the GPU: placed many times, with painted materials, lit by a sun, and water.
;; requires std/web/gpu std/web/scene/shader std/web/shade/color std/web/shade/noise std/web/scene/math std/web/scene/camera std/web/scene/light std/web/scene/materials

(const mesh-lit u32 0)

(const mesh-water u32 1)

(const mesh-cards u32 2)

(const material-land u32 20)

(const material-stone u32 21)

(const material-skin u32 22)

;; A placement's tint has alpha 0 for a small thing, such as grass.
(struct placement [m (array f32 12)] [tint u32])

(define (placement-at! [p (ptr placement)] [x f64] [y f64] [z f64] [yaw f64] [scale f64] [tint u32]))

;; A colour in a vertex or a placement is the bytes r, g, b, a.
(define (mesh-color [c u32] [sway f64]) -> u32)

(struct meshes
  [items (ptr mesh)] [count u32] [capacity u32]
  [frame u32]
  [shader lit-shader]
  [paints materials]
  [atlas-sampler u32] [atlas u32] [shadow-sampler u32] [shadow-maps u32] [heights u32] [blank-heights u32]
  [frame-uniforms u32] [cascade-uniforms u32] [frame-group u32] [surface-group u32] [shadow-group u32]
  [shadow-pipeline u32] [shadow-cards-pipeline u32] [lit-pipeline u32] [cards-pipeline u32] [water-pipeline u32]
  [heights-width f64] [heights-height f64])

(define (meshes-init! [r (ptr meshes)] [atlas string]))

(define (meshes-heights! [r (ptr meshes)] [texture u32] [width f64] [height f64]))

(define (meshes-add! [r (ptr meshes)] [kind u32]) -> u32)

(define (meshes-casts! [r (ptr meshes)] [mesh u32] [casts bool]))

(define (meshes-set! [r (ptr meshes)] [mesh u32] [vertices (ptr buffer)] [indices (ptr buffer)]))

(define (meshes-place! [r (ptr meshes)] [mesh u32] [places (ptr placement)] [count u32]))

(define (meshes-submit! [r (ptr meshes)] [mesh u32] [p (ptr placement)]))

(define (meshes-frame! [r (ptr meshes)] [c (ptr camera)] [l (ptr light)]))

(define (meshes-shadow-maps [r (ptr meshes)]) -> u32)

(define (meshes-shadow-pass! [r (ptr meshes)] [cascade u32]))

(define (meshes-pass! [r (ptr meshes)]))

;; ==== private ====

(const shadow-size u32 2048)

(const near-radius f64 28.0)

(const far-radius f64 120.0)

(define (clamp-unit [x f64]) -> f64)

(struct mesh
  [kind u32] [casts bool]
  [vertices u32] [vertex-room u32]
  [indices u32] [index-room u32]
  [places u32] [place-room u32]
  [index-count u32] [place-count u32]
  [submitted buffer] [submit-frame u32]
  [submits u32] [submit-room u32] [submit-count u32])

(define (shadows! [r (ptr meshes)]))

(define (room! [id (ptr u32)] [room (ptr u32)] [n u32] [usage u32]))

(define (submissions! [r (ptr meshes)]))

(define (draw-all! [r (ptr meshes)] [kind u32] [shadows bool]))

(define (cascade! [matrix (ptr f32)] [l (ptr light)] [sx f64] [sy f64] [sz f64] [radius f64]))
