;; std/web/scene/camera
;; The camera of a 3D scene: where it is, how it is turned, and what it sees.
;; requires std/web/scene/math

(struct camera [x f64] [y f64] [z f64] [yaw f64] [pitch f64] [near f64] [far f64] [projection (array f32 16)])

(define (camera-perspective! [c (ptr camera)] [fov f64] [aspect f64] [near f64] [far f64]))

(define (camera-axes! [c (ptr camera)] [right (ptr f32)] [up (ptr f32)] [forward (ptr f32)]))

(define (camera-half-width [c (ptr camera)]) -> f64)

(define (camera-half-height [c (ptr camera)]) -> f64)
