;; std/web/scene/models
;; Models and figures: things made of parts that move on joints.
;; requires std/web/game/vector std/web/scene/math std/web/scene/meshes compiler/macros

(struct part [mesh u32] [parent i32] [jx f64] [jy f64] [jz f64] [yaw f64] [pitch f64] [roll f64])

(struct model [parts (vector part)])

(define (model-add! [m (ptr model)] [mesh u32] [parent i32] [jx f64] [jy f64] [jz f64] [yaw f64] [pitch f64] [roll f64]) -> u32)

(define (model-count [m (ptr model)]) -> u32)

(struct joint [yaw f64] [pitch f64] [roll f64])

(struct figure
  [model (ptr model)]
  [x f64] [y f64] [z f64] [yaw f64] [pitch f64] [roll f64] [scale f64]
  [joints (vector joint)]
  [placed (vector placement)])

(define (figure-init! [f (ptr figure)] [m (ptr model)]))

(define (figure-free! [f (ptr figure)]))

(define (figure-joint [f (ptr figure)] [part u32]) -> (ptr joint))

(define (figure-pose! [f (ptr figure)] [part u32] [yaw f64] [pitch f64] [roll f64]))

(define (figure-turn! [f (ptr figure)] [part u32] [yaw f64] [pitch f64] [roll f64]))

(define (figure-rest! [f (ptr figure)]))

(define (figure-part-matrix [f (ptr figure)] [part u32]) -> (ptr f32))

(define (figure-part-position [f (ptr figure)] [part u32] [out (ptr f64)]))

(define (figure-submit! [f (ptr figure)] [r (ptr meshes)] [tint u32]))

(define (swing! [f (ptr figure)] [part u32] [phase f64] [amount f64]))

(define (walk! [f (ptr figure)] [hip u32] [knee u32] [phase f64] [swing f64] [bend f64]))

(define (bob [phase f64] [amount f64]) -> f64)

(define-syntax define-model)

(define (look-at! [f (ptr figure)] [part u32] [x f64] [y f64] [z f64] [limit f64]))

(define (ease-joint! [f (ptr figure)] [part u32] [yaw f64] [pitch f64] [roll f64] [rate f64]))

;; ==== private ====

(define (derived [name datum] [rest string]) -> datum)

(define (define-model-form [stx datum]) -> datum)
