;; std/web/scene/post
;; The picture pass: the textures a scene is drawn into, made into the picture on the screen.
;; requires std/web/gpu std/web/scene/shader std/web/shade/screen std/web/shade/color std/web/scene/camera

(struct post
  [width u32] [height u32] [scale u32]
  [color u32] [normals u32] [depth u32]
  [shader picture-shader] [uniforms u32] [group u32]
  [pipelines (array u32 10)])

(define (post-init! [p (ptr post)] [width u32] [height u32] [scale u32]))

(define (post-color [p (ptr post)]) -> u32)

(define (post-normals [p (ptr post)]) -> u32)

(define (post-depth [p (ptr post)]) -> u32)

(define (post-width [p (ptr post)]) -> u32)

(define (post-height [p (ptr post)]) -> u32)

(define (post-draw! [p (ptr post)] [kind u32] [target u32] [format u32] [c (ptr camera)]
                    [outline f64] [exposure f64]))

;; ==== private ====

(define (pipeline! [p (ptr post)] [format u32]) -> u32)
