;; std/web/scene/scene
;; A 3D scene drawn by the GPU: meshes lit by a sun, under a sky, made into a picture.
;; requires std/web/gpu std/web/scene/camera std/web/scene/light std/web/scene/meshes std/web/scene/sky std/web/scene/post std/web/scene/particles

(struct scene [meshes meshes] [dome sky] [picture post] [motes particles] [outline f64] [exposure f64])

(define (scene-init! [s (ptr scene)] [atlas string] [width u32] [height u32] [scale u32]))

(define (scene-meshes [s (ptr scene)]) -> (ptr meshes))

(define (scene-emit! [s (ptr scene)] [emitter u32] [x f64] [y f64] [z f64] [kind u32] [strength f64]))

(define (scene-draw! [s (ptr scene)] [kind u32] [target u32] [format u32] [c (ptr camera)] [l (ptr light)]))
