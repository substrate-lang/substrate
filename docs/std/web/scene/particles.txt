;; std/web/scene/particles
;; Particles: smoke, fireflies and splashes, moved by a compute shader and drawn as soft discs.
;; requires std/web/gpu std/web/scene/shader std/web/shade/noise std/web/scene/math std/web/scene/camera std/web/scene/light

(const particles-smoke u32 0)

(const particles-fireflies u32 1)

(const particles-splash u32 2)

(struct particles
  [stepper puffs-shader] [drawer motes-shader]
  [buffer u32] [step-uniforms u32] [draw-uniforms u32] [step-group u32] [draw-group u32]
  [step-pipeline u32] [draw-pipeline u32]
  [world puffs-world]
  [frame u32])

(define (particles-init! [p (ptr particles)]))

(define (particles-emit! [p (ptr particles)] [emitter u32] [x f64] [y f64] [z f64] [kind u32] [strength f64]))

(define (particles-step! [p (ptr particles)] [c (ptr camera)] [l (ptr light)]))

(define (particles-pass! [p (ptr particles)]))

;; ==== private ====

;; The shaders have these numbers written in: emitter e owns particles 64e to 64e + 63.
(const emitter-count u32 16)

(const slots u32 64)
