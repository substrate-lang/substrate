;; std/web/scene/math
;; The mathematics a 3D game needs that wasm has no instructions for: trigonometry, matrices.

(define pi f64)

(define (radians [degrees f64]) -> f64)

(define (clamp [x f64] [low f64] [high f64]) -> f64)

(define (sin [x f64]) -> f64)

(define (cos [x f64]) -> f64)

(define (tan [x f64]) -> f64)

(define (atan2 [y f64] [x f64]) -> f64)

(define (asin [x f64]) -> f64)

(define (acos [x f64]) -> f64)

(define (angle-between [a f64] [b f64]) -> f64)

(define (mat4-identity! [out (ptr f32)]))

(define (mat4-perspective! [out (ptr f32)] [fov f64] [aspect f64] [near f64] [far f64]))

(define (mat4-orthographic! [out (ptr f32)] [left f64] [right f64] [bottom f64] [top f64] [near f64] [far f64]))

(define (mat4-look! [out (ptr f32)] [x f64] [y f64] [z f64] [tx f64] [ty f64] [tz f64]))

(define (mat4-view! [out (ptr f32)] [x f64] [y f64] [z f64] [yaw f64] [pitch f64]))

(define (mat4-multiply! [out (ptr f32)] [a (ptr f32)] [b (ptr f32)]))

(define (mat4-transform! [out (ptr f64)] [m (ptr f32)] [x f64] [y f64] [z f64]))

(define (mat34-identity! [out (ptr f32)]))

;; Row by row: element (row, column) is at row * 4 + column.
(define (mat34-pose! [out (ptr f32)] [x f64] [y f64] [z f64] [yaw f64] [pitch f64] [roll f64] [scale f64]))

(define (mat34-multiply! [out (ptr f32)] [a (ptr f32)] [b (ptr f32)]))

(define (mat34-point! [out (ptr f64)] [m (ptr f32)] [x f64] [y f64] [z f64]))

(define (mat34-direction! [out (ptr f64)] [m (ptr f32)] [x f64] [y f64] [z f64]))

(define (mat34-yaw [m (ptr f32)]) -> f64)

(define (mat34-pitch [m (ptr f32)]) -> f64)

;; ==== private ====

(define (atan-unit [t f64]) -> f64)

;; Column by column, as WGSL reads a mat4x4f.
(define (at [m (ptr f32)] [row u32] [column u32]) -> f64)

(define (put! [m (ptr f32)] [row u32] [column u32] [value f64]))

(define (view-from-axes! [out (ptr f32)] [x f64] [y f64] [z f64]
                         [rx f64] [ry f64] [rz f64] [ux f64] [uy f64] [uz f64] [fx f64] [fy f64] [fz f64]))
