;; std/web/scene/light
;; The light of a 3D scene: the sun, the ambient light, the sky, the fog and the shadows.

(struct light
  [sun-x f64] [sun-y f64] [sun-z f64]
  [sun u32] [ambient u32] [sky u32]
  [fog-near f64] [fog-far f64]
  [mist-height f64] [mist f64]
  [time f64]
  [shadow-x f64] [shadow-y f64] [shadow-z f64] [shadow-radius f64])

(define (light-init! [l (ptr light)]))
