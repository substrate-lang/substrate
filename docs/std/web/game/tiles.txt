;; std/web/game/tiles
;; Tile maps for 2D games: levels of square tiles, bodies that move through them, a camera.
;; requires std/web/game/vector std/web/game/draw

(struct tilemap
  [columns i32] [rows i32]
  [size i32]
  [tiles (ptr u8)]
  [solid (array u8 256)])

(define (tilemap-init! [m (ptr tilemap)] [columns i32] [rows i32] [size i32]))

(define (tilemap-free! [m (ptr tilemap)]))

(define (tile-set! [m (ptr tilemap)] [column i32] [row i32] [tile u32]))

(define (tile-at [m (ptr tilemap)] [column i32] [row i32]) -> u32)

(define (tile-solid! [m (ptr tilemap)] [tile u32]))

(define (solid-at? [m (ptr tilemap)] [column i32] [row i32]) -> bool)

(struct body
  [x f64] [y f64] [width f64] [height f64]
  [vx f64] [vy f64]
  [on-ground bool] [hit-wall bool] [hit-ceiling bool])

(define (move-body! [m (ptr tilemap)] [b (ptr body)]))

(define (bodies-overlap? [a (ptr body)] [b (ptr body)]) -> bool)

(struct camera [x f64] [y f64])

(define (camera-follow! [c (ptr camera)] [x f64] [y f64] [screen-width f64] [screen-height f64] [m (ptr tilemap)]))

(define (draw-tilemap [m (ptr tilemap)] [sheet u32] [c (ptr camera)] [screen-width f64] [screen-height f64]))

(struct text-grid [columns i32] [rows i32] [lines (vector string)])

(define (text-grid-read! [g (ptr text-grid)] [text string]))

(define (grid-char [g (ptr text-grid)] [column i32] [row i32]) -> u32)

(define (text-grid-free! [g (ptr text-grid)]))

;; ==== private ====

(define (inside? [m (ptr tilemap)] [column i32] [row i32]) -> bool)

(define (cell-of [m (ptr tilemap)] [p f64]) -> i32)

(define (edge) -> f64)

(define (solid-column? [m (ptr tilemap)] [b (ptr body)] [x f64]) -> bool)

(define (solid-row? [m (ptr tilemap)] [b (ptr body)] [y f64]) -> bool)

(define (move-x! [b (ptr body)] [m (ptr tilemap)] [distance f64]))

(define (move-y! [b (ptr body)] [m (ptr tilemap)] [distance f64]))

(define (clamp [v f64] [low f64] [high f64]) -> f64)
