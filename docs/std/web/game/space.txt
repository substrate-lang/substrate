;; std/web/game/space
;; Space: a grid of cells over the world, for what is near, in the way, or hit by a line.
;; requires std/web/game/trap std/web/game/vector std/web/game/store std/web/scene/math compiler/macros

(variant shape
  [disc [r f64] [height f64]]
  [box [half-x f64] [half-z f64] [yaw f64] [height f64]]
  [walkway [half-x f64] [half-z f64] [yaw f64] [deck f64] [rails bool]])

(struct entry [shape shape] [kind u32] [id u32] [x f64] [y f64] [z f64] [cell i32] [slot u32])

(struct space
  [cell f64] [columns i32] [rows i32]
  [cells (ptr (vector (handle entry)))]
  [entries (store entry)])

(define (space-init! [s (ptr space)] [width f64] [depth f64] [cell f64]))

(define (space-free! [s (ptr space)]))

(define (space-count [s (ptr space)]) -> u32)

(define (space-add! [s (ptr space)] [shape shape] [kind u32] [id u32] [x f64] [y f64] [z f64]) -> (handle entry))

(define (space-move! [s (ptr space)] [h (handle entry)] [x f64] [y f64] [z f64]))

(define (space-remove! [s (ptr space)] [h (handle entry)]))

(define (space-entry [s (ptr space)] [h (handle entry)]) -> (ptr entry))

(define (kinds-all) -> u32)

(define (kind-mask [k u32]) -> u32)

(define (space-gather! [s (ptr space)] [x f64] [z f64] [r f64] [kinds u32] [results (ptr (vector (handle entry)))]))

(define (space-nearest [s (ptr space)] [x f64] [z f64] [r f64] [kinds u32]) -> (handle entry))

(define (space-touching! [s (ptr space)] [x f64] [z f64] [y f64] [r f64] [height f64] [kinds u32]
                         [results (ptr (vector (handle entry)))]))

(define (space-ray [s (ptr space)] [x f64] [y f64] [z f64] [dx f64] [dy f64] [dz f64] [reach f64] [kinds u32]
                   [distance (ptr f64)]) -> (handle entry))

;; ==== private ====

(define (column-of [s (ptr space)] [x f64]) -> i32)

(define (row-of [s (ptr space)] [z f64]) -> i32)

(define (cell-of [s (ptr space)] [x f64] [z f64]) -> i32)

(define (file! [s (ptr space)] [e (ptr entry)] [h (handle entry)] [cell i32]))

(define (unfile! [s (ptr space)] [e (ptr entry)]))

(define (cells-around-form [stx datum]) -> datum)

(define-syntax cells-around)

(define (wanted? [e (ptr entry)] [kinds u32]) -> bool)

(define (touches? [e (ptr entry)] [x f64] [y f64] [z f64] [r f64] [height f64]) -> bool)

(define (inside? [e (ptr entry)] [x f64] [y f64] [z f64]) -> bool)
