;; std/web/game/format
;; Formatting text, for games: numbers and strings put into text by a pattern.
;; requires compiler/macros

;; ---- the forms ----

(define-syntax format-into)

(define-syntax buffer-add-format!)

(define-syntax format-string)

(define-syntax print-format)

;; ==== private ====

;; ---- where the text goes ----

(struct format-writer [bytes (ptr u8)] [room u32] [buffer (ptr buffer)] [own buffer] [length u32] [cut bool])

(define (format-start! [w (ptr format-writer)] [into (slice u8)]))

(define (format-start-buffer! [w (ptr format-writer)] [b (ptr buffer)]))

(define (format-start-own! [w (ptr format-writer)]))

(define (format-add! [w (ptr format-writer)] [part string]))

(define (format-fill! [w (ptr format-writer)] [byte u32] [count u32]))

(define (format-finish! [w (ptr format-writer)]) -> u32)

(define (format-finish-text! [w (ptr format-writer)]) -> string)

(define (format-finish-string! [w (ptr format-writer)]) -> string)

(define (format-finish-print! [w (ptr format-writer)]))

;; ---- the directives ----

(const flag-left u32 1)

(const flag-zeros u32 2)

(define (format-padded! [w (ptr format-writer)] [negative bool] [text string] [width u32] [flags u32]))

(define (digits! [into (slice u8)] [n u64] [base u64]) -> string)

(define (format-unsigned! [w (ptr format-writer)] [n u64] [base u32] [width u32] [flags u32]))

(define (format-integer! [w (ptr format-writer)] [n i64] [width u32] [flags u32]))

(define (format-text! [w (ptr format-writer)] [s string] [width u32] [flags u32]))

(define (format-character! [w (ptr format-writer)] [c u32]))

(define (float-digits! [text (ptr u8)] [x f64] [decimals i32] [zero (ptr bool)]) -> u32)

(define (format-float! [w (ptr format-writer)] [x f64] [decimals i32] [width u32] [flags u32]))

;; ---- the forms ----

(define (fail [where datum] [message datum]) -> datum)

(define (literal [d datum]) -> datum)

(define (number [n i64]) -> datum)

(define (byte-at [s string] [i u32]) -> u32)

(define (digit? [c u32]) -> bool)

(define (statements [pattern datum] [arguments datum]) -> datum)

(define (format-into-form [s datum]) -> datum)

(define (buffer-add-format-form [s datum]) -> datum)

(define (format-string-form [s datum]) -> datum)

(define (print-format-form [s datum]) -> datum)
