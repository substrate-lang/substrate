;; std/web/game/movers
;; Movers: things that move over the land among the things that stand on it.
;; requires std/web/game/vector std/web/game/store std/web/game/terrain std/web/game/space std/web/game/noise std/web/scene/math

;; heading is a yaw: 0 faces +z, and it turns towards +x as it grows.
(struct mover
  [x f64] [y f64] [z f64]
  [vx f64] [vz f64] [vy f64]
  [heading f64] [speed f64]
  [radius f64] [height f64]
  [on-ground bool] [in-water f64]
  [gravity f64] [slope-limit f64] [wade f64] [swims bool]
  [kind u32] [entry (handle entry)] [entered bool]
  [touching (vector (handle entry))])

(define (mover-init! [m (ptr mover)] [x f64] [y f64] [z f64] [radius f64] [height f64]))

(define (mover-enter! [m (ptr mover)] [s (ptr space)] [kind u32] [id u32]))

(define (mover-leave! [m (ptr mover)] [s (ptr space)]))

(define (mover-jump! [m (ptr mover)] [speed f64]))

(define (mover-aim! [m (ptr mover)]))

(define (mover-distance [m (ptr mover)] [x f64] [z f64]) -> f64)

(define (mover-bearing [m (ptr mover)] [x f64] [z f64]) -> f64)

(define (steer-toward! [m (ptr mover)] [x f64] [z f64] [turn f64]))

(define (steer-away! [m (ptr mover)] [x f64] [z f64] [turn f64]))

(define (steer-wander! [m (ptr mover)] [seed u32] [time f64] [turn f64]))

(define (steer-arrive! [m (ptr mover)] [x f64] [z f64] [slow f64] [top f64]))

(define (steer-apart! [m (ptr mover)] [s (ptr space)] [kinds u32] [distance f64] [turn f64]))

(define (steer-clear! [m (ptr mover)] [s (ptr space)] [kinds u32] [turn f64]))

(define (mover-step! [m (ptr mover)] [t (ptr terrain)] [s (ptr space)] [water-level f64]))

;; ==== private ====

(define (turn-toward! [m (ptr mover)] [bearing f64] [turn f64]))

(define (own? [m (ptr mover)] [h (handle entry)]) -> bool)

(define (blocked? [m (ptr mover)] [s (ptr space)] [kinds u32] [x f64] [z f64]) -> bool)

(define (resolve! [m (ptr mover)] [e (ptr entry)]))

(define (swim-step! [m (ptr mover)] [t (ptr terrain)] [s (ptr space)] [water-level f64]))

(define (shapes! [m (ptr mover)] [s (ptr space)]))

(define (walk-step! [m (ptr mover)] [t (ptr terrain)] [s (ptr space)] [water-level f64]))
