;; std/web/game/input
;; Keyboard and mouse state for games.

(const key-none u32 0)

(const key-up u32 1)

(const key-down u32 2)

(const key-left u32 3)

(const key-right u32 4)

(const key-space u32 5)

(const key-enter u32 6)

(const key-escape u32 7)

(const key-shift u32 8)

(const mouse-left u32 0)

(const mouse-middle u32 1)

(const mouse-right u32 2)

(struct input [held (array u8 128)] [pressed (array u8 128)] [released (array u8 128)]
  [buttons-held (array u8 3)] [buttons-pressed (array u8 3)] [buttons-released (array u8 3)]
  [dx f64] [dy f64] [locked bool])

(define (held? [in (ptr input)] [code u32]) -> bool)

(define (pressed? [in (ptr input)] [code u32]) -> bool)

(define (released? [in (ptr input)] [code u32]) -> bool)

(define (mouse-held? [in (ptr input)] [button u32]) -> bool)

(define (mouse-pressed? [in (ptr input)] [button u32]) -> bool)

(define (mouse-released? [in (ptr input)] [button u32]) -> bool)

(define (mouse-dx [in (ptr input)]) -> f64)

(define (mouse-dy [in (ptr input)]) -> f64)

(define (mouse-locked? [in (ptr input)]) -> bool)

(define (key-code [name string]) -> u32)

(define (input-key-down! [in (ptr input)] [code u32]))

(define (input-key-up! [in (ptr input)] [code u32]))

(define (input-mouse-down! [in (ptr input)] [button u32]))

(define (input-mouse-up! [in (ptr input)] [button u32]))

(define (input-mouse-move! [in (ptr input)] [dx f64] [dy f64]))

(define (input-mouse-lock! [in (ptr input)] [locked bool]))

(define (input-end-step! [in (ptr input)]))

(define (input-clear! [in (ptr input)]))

;; ==== private ====

(define (valid? [code u32]) -> bool)
