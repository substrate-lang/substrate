;; std/web/game/voxels
;; Worlds made of blocks: a grid, bodies that move through it, rays, and its meshes.
;; requires std/web/game/trap

(struct voxels [width i32] [height i32] [depth i32] [blocks (ptr u8)])

(define (voxels-init! [v (ptr voxels)] [width i32] [height i32] [depth i32]))

(define (voxels-free! [v (ptr voxels)]))

(define (block-at [v (ptr voxels)] [x i32] [y i32] [z i32]) -> u8)

(define (block-set! [v (ptr voxels)] [x i32] [y i32] [z i32] [kind u8]))

(define (solid-at? [v (ptr voxels)] [x i32] [y i32] [z i32]) -> bool)

(struct body [x f64] [y f64] [z f64] [vx f64] [vy f64] [vz f64] [width f64] [height f64] [depth f64]
  [on-ground bool] [hit-wall bool] [hit-ceiling bool])

(define (body-free? [v (ptr voxels)] [b (ptr body)] [dx f64] [dy f64] [dz f64]) -> bool)

(define (move-body! [v (ptr voxels)] [b (ptr body)]))

(struct hit [x i32] [y i32] [z i32] [face i32] [distance f64])

(define (face-normal [face i32] [axis i32]) -> i32)

;; Amanatides and Woo's walk from block to block.
(define (ray-hit! [v (ptr voxels)] [x f64] [y f64] [z f64] [dx f64] [dy f64] [dz f64] [reach f64] [h (ptr hit)]) -> bool)

(struct tiles [top u32] [side u32] [bottom u32])

(define (mesh-blocks! [v (ptr voxels)] [x0 i32] [y0 i32] [z0 i32] [w i32] [h i32] [d i32]
                      [table (ptr tiles)] [tile u32] [vertices (ptr buffer)] [indices (ptr buffer)]) -> u32)

;; ==== private ====

(define (inside? [v (ptr voxels)] [x i32] [y i32] [z i32]) -> bool)

(define (place [v (ptr voxels)] [x i32] [y i32] [z i32]) -> u32)

(define small f64)

(define (cell [c f64]) -> i32)

(define (blocked? [v (ptr voxels)] [b (ptr body)]) -> bool)

(define (face-axes! [axes (ptr i32)] [face u32]))

(define (openness [v (ptr voxels)] [x i32] [y i32] [z i32] [nx i32] [ny i32] [nz i32]
                  [sx i32] [sy i32] [sz i32] [tx i32] [ty i32] [tz i32]) -> i32)

(define (face! [v (ptr voxels)] [x i32] [y i32] [z i32] [face u32] [f (ptr i32)] [tile-at u32] [tile u32]
               [ox i32] [oy i32] [oz i32] [vertices (ptr buffer)] [indices (ptr buffer)]))
