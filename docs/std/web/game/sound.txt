;; std/web/game/sound
;; Sound for games: sound files, and short synthesized tones.
;; requires std/web/game/vector

;; As host/web/audio.mjs numbers them.
(const square-wave u32 0)

(const sawtooth-wave u32 1)

(const triangle-wave u32 2)

(const sine-wave u32 3)

(struct tone [wave u32] [from f64] [to f64] [ms f64] [volume f64])

(define (play-tone [wave u32] [from f64] [to f64] [ms f64] [volume f64]))

(define (tones-waiting) -> u32)

(define (tone-at [i u32]) -> (ptr tone))

(define (tones-clear!))

(struct sound-file [id u32] [path string])

(define (load-sound [path string]) -> u32)

(define (sound-files-waiting) -> u32)

(define (sound-file-at [i u32]) -> (ptr sound-file))

(define (sound-files-clear!))

(struct sound-play [sound u32] [volume f64])

(define (play-sound [sound u32] [volume f64]))

(define (sounds-waiting) -> u32)

(define (sound-at [i u32]) -> (ptr sound-play))

(define (sounds-clear!))

(const music-same u32 0)

(const music-start u32 1)

(const music-stop u32 2)

(define (play-music [sound u32] [volume f64]))

(define (stop-music))

(define (music-change) -> u32)

(define (music-sound) -> u32)

(define (music-volume) -> f64)

(define (music-change-clear!))

;; ==== private ====

(define tones (array tone 16))

(define tone-count u32)

(define files (vector sound-file))

(define next-sound u32)

(define plays (array sound-play 16))

(define play-count u32)

(define change u32)

(define wanted u32)

(define level f64)
