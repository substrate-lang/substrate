;; std/web/game/terrain
;; Land as a height map: meshed for the scene, asked for its height and slope, walked on.
;; requires std/web/gpu std/web/scene/meshes

(struct terrain [columns i32] [rows i32] [spacing f64] [heights (ptr f32)])

(define (terrain-init! [t (ptr terrain)] [columns i32] [rows i32] [spacing f64]))

(define (terrain-free! [t (ptr terrain)]))

(define (terrain-set! [t (ptr terrain)] [column i32] [row i32] [height f64]))

(define (terrain-height [t (ptr terrain)] [x f64] [z f64]) -> f64)

(define (terrain-slope [t (ptr terrain)] [x f64] [z f64]) -> f64)

(define (terrain-mesh! [t (ptr terrain)] [vertices (ptr buffer)] [indices (ptr buffer)] [tile-x f64] [tile-y f64] [material u32]
                       [color-of (fn (f64 f64 f64 f64) -> u32)]))

(define (terrain-texture! [t (ptr terrain)]) -> u32)

(struct walker [x f64] [y f64] [z f64] [vy f64] [on-ground bool])

(define (walker-move! [t (ptr terrain)] [w (ptr walker)] [dx f64] [dz f64]))

;; ==== private ====

(define (clamp-index [i i32] [n i32]) -> i32)

(define (height-at [t (ptr terrain)] [column i32] [row i32]) -> f64)

(define (half [x f32]) -> u32)
