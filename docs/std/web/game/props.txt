;; std/web/game/props
;; Props: the things that stand in a world and do not move.
;; requires std/web/game/vector std/web/game/store std/web/game/space std/web/scene/math std/web/scene/camera std/web/scene/meshes std/web/scene/models

(struct prop
  [model (ptr model)] [x f64] [y f64] [z f64] [yaw f64] [scale f64] [tint u32]
  [place placement]
  [entry (handle entry)] [shaped bool]
  [cell i32] [near bool] [slot u32])

(struct props
  [space (ptr space)]
  [cellsize f64] [columns i32] [rows i32]
  [cells (ptr cell)]
  [items (store prop)]
  [handles (vector (handle prop))]
  [figure figure] [figured bool]
  [making bool] [making-cell i32]
  [near-column i32] [near-row i32] [near-reach f64])

(define (props-init! [p (ptr props)] [s (ptr space)]))

(define (props-free! [p (ptr props)]))

(define (props-count [p (ptr props)]) -> u32)

(define (props-add! [p (ptr props)] [m (ptr model)] [x f64] [y f64] [z f64] [yaw f64] [scale f64] [tint u32]) -> (handle prop))

(define (props-shape! [p (ptr props)] [h (handle prop)] [shape shape] [kind u32]))

(define (props-get [p (ptr props)] [h (handle prop)]) -> (ptr prop))

(define (props-of-entry [p (ptr props)] [e (ptr entry)]) -> (ptr prop))

(define (props-remove! [p (ptr props)] [h (handle prop)]))

(define (props-near! [p (ptr props)] [x f64] [z f64] [reach f64] [make (fn ((ptr props) f64 f64 f64 f64))]))

(define (props-submit! [p (ptr props)] [r (ptr meshes)] [c (ptr camera)] [far f64]) -> u32)

;; ==== private ====

(struct cell
  [fixed (vector (handle prop))] [near (vector (handle prop))]
  [made bool]
  [low f64] [high f64])

(define (cell-of [p (ptr props)] [x f64] [z f64]) -> i32)

(define (unfile! [p (ptr props)] [pr (ptr prop)]))

(define (seen? [c (ptr camera)] [right (ptr f32)] [up (ptr f32)] [forward (ptr f32)] [half-w f64] [half-h f64]
               [x f64] [y f64] [z f64] [radius f64] [far f64]) -> bool)

(define (submit-prop! [p (ptr props)] [pr (ptr prop)] [r (ptr meshes)]))
