;; std/web/game/game
;; The game loop: what turns three functions into a game in a web page.
;; requires std/web/web std/web/gpu std/web/game/input host/web/audio std/web/game/sound std/web/game/draw std/web/game/trap compiler/macros

;; ---- the `game` form ----

(define-syntax game)

;; ==== private ====

;; ---- the loop's own state ----

(struct game-loop
  [input input]
  [step-ms f64]
  [timed bool]
  [last-time f64]
  [pending f64]
  [frame u32]
  [started bool]
  [music i32]
  [surface-node u32]
  [surface u32]
  [wants-pointer bool]
  [lock-requests u32]
  [stopped bool]
  [problem string])

(const max-steps u32 5)

(define (game-loop-start! [l (ptr game-loop)] [steps-per-second f64]))

(define (game-loop-steps! [l (ptr game-loop)] [now f64]) -> u32)

(define (game-loop-first-frame? [l (ptr game-loop)]) -> bool)

(define (game-loop-running? [l (ptr game-loop)]) -> bool)

(define (game-loop-stopped? [l (ptr game-loop)]) -> bool)

(define (game-loop-trapped! [l (ptr game-loop)] [why string]))

(define (game-loop-problem [l (ptr game-loop)]) -> string)

(define (game-loop-step-done! [l (ptr game-loop)]))

(define (game-loop-drawing? [l (ptr game-loop)]) -> bool)

(define (game-loop-surface [l (ptr game-loop)]) -> u32)

(define (game-loop-gpu-event! [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))

(define (game-loop-frame-done! [l (ptr game-loop)]))

(define (game-loop-key! [l (ptr game-loop)] [name string] [down bool]) -> bool)

(define (game-loop-blur! [l (ptr game-loop)]))

(define (game-loop-mouse-down! [l (ptr game-loop)] [button i32]))

(define (game-loop-mouse-up! [l (ptr game-loop)] [button i32]))

(define (game-loop-mouse-moved! [l (ptr game-loop)] [dx i32] [dy i32]))

(define (game-loop-pointer! [l (ptr game-loop)] [locked bool]))

;; ---- the `game` form ----

(define (game-form [s datum]) -> datum)

(define (option? [x datum]) -> bool)

(define (game-form-with [s datum] [world datum] [width datum] [height datum] [renderer datum] [pointer bool]
                        [init datum] [step datum] [draw datum]) -> datum)
