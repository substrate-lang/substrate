;; std/log/log
;; Logs after Go's log/slog: an entry is a level, a message and attributes, written as one line of key=value text or
;; JSON when its level is at least the logger's. A text value is quoted when it needs to be; a byte that would break the
;; line is escaped as JSON escapes it, and one that is not UTF-8 is written as U+FFFD.
;; requires host/modules/clock std/encoding/utf8

(const log-debug i32 -4)

(const log-info i32 0)

(const log-warn i32 4)

(const log-error i32 8)

;; level is the least level written, log-info when 0; json writes JSON lines; lines go to write when it is set, or else
;; to fd, stderr when 0.
(struct logger [level i32] [json bool] [fd i32] [write (fn ((ptr logger) (slice u8)))])

;; An entry that log-begin! does not start, for its level, takes no attributes and writes nothing.
(struct log-entry [logger (ptr logger)] [line buffer] [valid bool])

(define log-default logger)

(define (log-string! [e (ptr log-entry)] [key string] [value string]))

(define (log-int! [e (ptr log-entry)] [key string] [value i64]))

;; log-float! writes a value that is not a finite number as null in JSON, and in text as NaN, +Inf or -Inf.
(define (log-float! [e (ptr log-entry)] [key string] [value f64]))

(define (log-bool! [e (ptr log-entry)] [key string] [value bool]))

;; log-begin! starts an entry, with the time, the level and the message, and returns #f when the level is below the
;; logger's.
(define (log-begin! [e (ptr log-entry)] [l (ptr logger)] [level i32] [message string]) -> bool)

(define (log-end! [e (ptr log-entry)]))

(define (log! [l (ptr logger)] [level i32] [message string]))

;; ==== private ====

(define (level-name [level i32]) -> string)

(define (two! [b (ptr buffer)] [n i64]))

;; time! writes the time, in RFC 3339 to the millisecond in UTC.
(define (time! [b (ptr buffer)] [ms f64]))

(define (plain? [s string]) -> bool)

(define (quoted! [b (ptr buffer)] [s string]))

(define (key! [e (ptr log-entry)] [key string]))
