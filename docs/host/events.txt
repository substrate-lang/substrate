;; host/events
;; Events from the host. events-take gives the current event's bytes as host calls give them, events-take-text its
;; text and events-take-strings its list; event-handler! sets the handler for a source's untagged events, and a tagged
;; event goes to its tag's handler (compiler/substrate.sub). A failed event's value is one of the error codes here,
;; and its text is the reason.
;; Sources: dom 1, gpu 2, worker 3, audio 4, timer 5, fetch 6, websocket 7, stream 8, net 9, process 10, storage 11,
;; location 12, clipboard 13, files 14.

(const error-missing i32 1)

(const error-unreachable i32 2)

(const error-refused i32 3)

(const error-timeout i32 4)

(const error-invalid i32 5)

(const error-aborted i32 6)

(const error-denied i32 7)

(const error-in-use i32 8)

(const error-other i32 9)

(const error-not-granted i32 10)

(define (event-handler! [source u32] [f handler]))

(define (events-take) -> (slice u8))

(define (events-take-text) -> string)

(define (events-take-strings) -> (slice string))

(define (events-take! [b (ptr buffer)]) -> u32)

;; ==== private ====

(import take "events" "take" (i32 -> i32))

(define defaults (array handler 64))

(define (by-source [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))
