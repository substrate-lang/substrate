;; host/stream
;; Byte streams: the host module "stream", whose ids other modules give out. A read event's value is how many bytes
;; were read, or 0 at the end. A read that stream-take! posts holds no room: its event carries the bytes, at most
;; most of them, for events-take. A finished event comes once everything written before stream-finish has been sent.
;; requires host/events

(const stream-source u32 8)

(const stream-read u32 1)

(const stream-drained u32 2)

(const stream-failed u32 3)

(const stream-finished u32 4)

(define (stream-read! [id u32] [into (slice u8)]) -> i32)

(define (stream-take! [id u32] [most u32]) -> i32)

(define (stream-write [id u32] [b (slice u8)]) -> i32)

(define (stream-write-text [id u32] [s string]) -> i32)

(define (stream-finish [id u32]))

(define (stream-close [id u32]))

(define (stream-tag! [id u32] [t (ptr void)]))

;; A reader takes a stream's bytes into its buffer until the end or a failure, then calls finished.
(struct stream-reader [head event-head] [id u32] [bytes buffer] [failed i32] [chunk (array u8 4096)]
  [finished (fn ((ptr stream-reader)))])

(define (stream-read-all! [r (ptr stream-reader)] [id u32] [finished (fn ((ptr stream-reader)))]))

;; A pipe moves n bytes, or all of them when n is -1, from one stream to another, then closes the source and calls
;; finished. A source that ends before n bytes fails the pipe. The destination's owner calls stream-pipe-drained!
;; when that stream drains.
(struct stream-pipe [head event-head] [from u32] [to u32] [left i64] [finish bool] [waiting bool] [failed i32]
  [chunk (array u8 65536)] [finished (fn ((ptr stream-pipe)))])

(define (stream-pipe-start! [p (ptr stream-pipe)] [from u32] [to u32] [n i64] [finish bool]
                            [finished (fn ((ptr stream-pipe)))]))

(define (stream-pipe-drained! [p (ptr stream-pipe)]))

(define (pipe-heard [source u32] [kind u32] [id u32] [tag (ptr void)] [value i32]))
