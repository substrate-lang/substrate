;; host/web/storage
;; Keys and their bytes, kept by the browser: the host module "storage". A got event's value is 1 when the key was
;; found, and its bytes are the key's. A keys event's value is how many keys there are, and its bytes are the keys as
;; entries.

(const storage-source u32 11)

(const storage-got u32 1)

(const storage-done u32 2)

(const storage-keys u32 3)

(const storage-failed u32 4)

(define (storage-get [key string] [tag (ptr void)]) -> u32)

(define (storage-set [key string] [b (slice u8)] [tag (ptr void)]) -> u32)

(define (storage-set-text [key string] [text string] [tag (ptr void)]) -> u32)

(define (storage-remove [key string] [tag (ptr void)]) -> u32)

(define (storage-list [prefix string] [tag (ptr void)]) -> u32)

;; ==== private ====

(import get "storage" "get" (i32 i32 i32 -> i32))

(import set "storage" "set" (i32 i32 i32 i32 i32 -> i32))

(import remove "storage" "remove" (i32 i32 i32 -> i32))

(import list "storage" "list" (i32 i32 i32 -> i32))
