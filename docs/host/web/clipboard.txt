;; host/web/clipboard
;; The clipboard's text, used with the user's leave: the host module "clipboard". A text event's bytes are the text.

(const clipboard-source u32 13)

(const clipboard-text u32 1)

(const clipboard-written u32 2)

(const clipboard-failed u32 3)

(define (clipboard-read [tag (ptr void)] [err (ptr string)]) -> u32)

(define (clipboard-write [text string] [tag (ptr void)] [err (ptr string)]) -> u32)

;; ==== private ====

(import read "clipboard" "read" (i32 i32 -> i32))

(import write "clipboard" "write" (i32 i32 i32 i32 -> i32))
