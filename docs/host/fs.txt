;; host/fs
;; Files: the host module "fs", on paths and on open files. fs-list stores a directory's names
;; at into, and fs-read-link, fs-real-path, fs-make-temp-dir, fs-make-temp-file and fs-cwd a path, as host calls give
;; them. fs-stat fills an fs-info, and fs-lstat does too without following a link at the end of the path. fs-remove
;; removes a file, a link or an empty directory, or when recursive a directory and everything in it. The calls return
;; 0, or -1 when they fail, and store a failure's reason at err.

;; An fs-info's kind.
(const fs-file i32 1)

(const fs-directory i32 2)

(const fs-link i32 3)

(const fs-other i32 4)

;; modified is in milliseconds since 1970, and mode is the file's mode bits, its type's included.
(struct fs-info [kind i32] [size i64] [modified f64] [mode u32])

(define (fs-make-dir [path string] [err (ptr string)]) -> i32)

(define (fs-stat [path string] [i (ptr fs-info)] [err (ptr string)]) -> i32)

(define (fs-lstat [path string] [i (ptr fs-info)] [err (ptr string)]) -> i32)

(define (fs-list [path string] [into (ptr (slice string))] [err (ptr string)]) -> i32)

(define (fs-remove [path string] [recursive bool] [err (ptr string)]) -> i32)

(define (fs-rename [from string] [to string] [err (ptr string)]) -> i32)

(define (fs-read-link [path string] [into (ptr string)] [err (ptr string)]) -> i32)

(define (fs-real-path [path string] [into (ptr string)] [err (ptr string)]) -> i32)

;; fs-symlink makes a link at path to target, which needs all of -R and -W, as Deno's symlink does.
(define (fs-symlink [target string] [path string] [err (ptr string)]) -> i32)

(define (fs-copy [from string] [to string] [err (ptr string)]) -> i32)

(define (fs-chmod [path string] [mode u32] [err (ptr string)]) -> i32)

(define (fs-make-temp-dir [dir string] [prefix string] [suffix string] [into (ptr string)] [err (ptr string)]) -> i32)

(define (fs-make-temp-file [dir string] [prefix string] [suffix string] [into (ptr string)] [err (ptr string)]) -> i32)

(define (fs-cwd [into (ptr string)] [err (ptr string)]) -> i32)

;; Open files, read and written at a position. fs-open-file gives a handle opened as flags say, to read when they say
;; neither read nor write, or 0 when it fails; writing, appending, creating and truncating need -W, and reading -R.
;; fs-read-at gives the number of bytes it read, fewer at the end of the file, and fs-size the file's size, or -1 when
;; it fails. fs-read-stream and fs-write-stream make a handle a stream from start, which closes the handle when it
;; closes, or give 0 when they fail, closing it. fs-read-file stores a file's bytes, from malloc, at into, and
;; fs-write-file makes a file hold b.
(const fs-read-flag u32 1)

(const fs-write-flag u32 2)

(const fs-append-flag u32 4)

(const fs-truncate-flag u32 8)

(const fs-create-flag u32 16)

(const fs-create-new-flag u32 32)

(define (fs-open-file [path string] [flags u32] [err (ptr string)]) -> u32)

(define (fs-read-at [h u32] [into (slice u8)] [at i64] [err (ptr string)]) -> i32)

(define (fs-write-at [h u32] [b (slice u8)] [at i64] [err (ptr string)]) -> i32)

(define (fs-set-size [h u32] [n i64] [err (ptr string)]) -> i32)

(define (fs-sync [h u32] [err (ptr string)]) -> i32)

(define (fs-size [h u32] [err (ptr string)]) -> i64)

(define (fs-close-file [h u32] [err (ptr string)]) -> i32)

(define (fs-read-stream [h u32] [start i64] [tag (ptr void)] [err (ptr string)]) -> u32)

(define (fs-write-stream [h u32] [start i64] [tag (ptr void)] [err (ptr string)]) -> u32)

(define (fs-read-file [path string] [into (ptr (slice u8))] [err (ptr string)]) -> i32)

(define (fs-write-file [path string] [b (slice u8)] [err (ptr string)]) -> i32)
