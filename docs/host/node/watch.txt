;; host/node/watch
;; Watching files: the host module "watch", after Deno's watchFs. watch-start watches the paths, and everything under
;; them unless the options' recursive is #f, and returns the watch's number, or 0 with the reason at err. A changed
;; event's value is watch-create, watch-modify or watch-remove, and its bytes are the absolute paths it is about, as
;; entries; a rename comes as a remove and a create. Changes under the options' ignore paths are left out. A failed
;; event ends the watch.

(const watch-source u32 15)

(const watch-changed u32 1)

(const watch-failed u32 2)

(const watch-create i32 1)

(const watch-modify i32 2)

(const watch-remove i32 3)

(struct watch-options [recursive bool] [ignore (slice string)])

(define (watch-options-init! [o (ptr watch-options)]))

(define (watch-start [paths (slice string)] [o (ptr watch-options)] [tag (ptr void)] [err (ptr string)]) -> u32)

(define (watch-stop [w u32]))

;; ==== private ====

(import start "watch" "start" (i32 i32 i32 i32 i32 i32 i32 -> i32))

(import stop "watch" "stop" (i32 ->))
