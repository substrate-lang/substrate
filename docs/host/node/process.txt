;; host/node/process
;; Child programs: the host module "process", whose pipes are streams. process-spawn takes the arguments, and options,
;; or 0 for the defaults. The options' env strings, NAME=VALUE, are added to the parent's environment, or replace it
;; when clear is set. An exit event's value is the exit code, or 128 plus the signal. process-exec replaces this
;; program's process with another program, keeping its environment and standard streams; it returns only when that
;; fails. process-pid is the program's own process number.

(const process-source u32 10)

(const process-exit u32 1)

(const process-failed u32 2)

(const process-pipe i32 0)

(const process-ignore i32 1)

(const process-inherit i32 2)

(struct process-options [dir string] [env (slice string)] [clear bool] [stdin i32] [stdout i32] [stderr i32])

(define (process-options-init! [o (ptr process-options)]))

(define (process-spawn [program string] [args (slice string)] [o (ptr process-options)] [tag (ptr void)]
                       [err (ptr string)]) -> u32)

(define (process-exec [program string] [args (slice string)] [err (ptr string)]) -> i32)

(define (process-exec-at-end [program string] [args (slice string)] [err (ptr string)]) -> i32)

(define (process-restrict [grants (slice string)] [err (ptr string)]) -> i32)

(define (process-stdin [id u32]) -> u32)

(define (process-stdout [id u32]) -> u32)

(define (process-stderr [id u32]) -> u32)

(define (process-kill [id u32] [signal u32]))

(define (process-pid) -> i32)
