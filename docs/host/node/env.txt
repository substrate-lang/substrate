;; host/node/env
;; Environment variables: the host module "env". env-get stores a variable's value at value and returns #t, or returns
;; #f when it is not set or not granted, with the reason at err. env-set and env-delete change the program's
;; environment, which the programs it runs inherit, and return 0, or -1 with the reason at err. env-list stores the
;; variables -E grants at into, NAME=VALUE, in name order, as Deno's toObject does, and returns 0, or -1 without -E.

(define (env-get [name string] [value (ptr string)] [err (ptr string)]) -> bool)

(define (env-set [name string] [value string] [err (ptr string)]) -> i32)

(define (env-delete [name string] [err (ptr string)]) -> i32)

(define (env-list [into (ptr (slice string))] [err (ptr string)]) -> i32)

;; ==== private ====

(import get "env" "get" (i32 i32 i32 i32 -> i32))

(import put "env" "set" (i32 i32 i32 i32 i32 -> i32))

(import drop "env" "delete" (i32 i32 i32 -> i32))

(import listed "env" "list" (i32 i32 -> i32))
