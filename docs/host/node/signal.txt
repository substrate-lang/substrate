;; host/node/signal
;; Signals to the program: the host module "signal", after Deno's addSignalListener. signal-listen listens for the
;; signal named, such as "SIGINT" or "SIGTERM", and returns the listener's number, or 0 with the reason at err; while
;; any listener is there, the signal no longer ends the program, and a listener alone does not keep it running. A
;; received event comes each time the signal arrives. SIGKILL, SIGSTOP, SIGILL, SIGFPE and SIGSEGV cannot be listened
;; for, and a worker has no signals.

(const signal-source u32 16)

(const signal-received u32 1)

(define (signal-listen [name string] [tag (ptr void)] [err (ptr string)]) -> u32)

(define (signal-stop [id u32]))

;; ==== private ====

(import listen "signal" "listen" (i32 i32 i32 i32 -> i32))

(import stop "signal" "stop" (i32 ->))
