;; host/worker
;; Workers: this program's functions, each run on a thread that shares only messages. worker-start runs one of the
;; program's own functions with the bytes as its argument 0, and returns the worker's number, or 0 with the reason at
;; err. Options: a time limit in ms for each stretch of work and a memory limit in bytes, none at 0; output as stdout
;; and stderr events; grants, flags such as -R=DIR within the parent's, else the parent's own. Calling a module a thread
;; cannot have traps. A done event's bytes are the trap, if any; an ended event's value is the exit code, or -1 after a
;; trap, the time limit or a failure, with the reason as its bytes.

(const worker-source u32 3)

(const worker-message u32 1)

(const worker-stdout u32 2)

(const worker-stderr u32 3)

(const worker-done u32 4)

(const worker-ended u32 5)

(struct worker-options [time u32] [memory u32] [output bool] [grants (slice string)])

(define (worker-options-init! [o (ptr worker-options)]))

(define (worker-start [function (fn ())] [argument (slice u8)] [o (ptr worker-options)] [tag (ptr void)]
                      [err (ptr string)]) -> u32)

(define (worker-send [w u32] [b (slice u8)]) -> i32)

(define (worker-send-text [w u32] [s string]) -> i32)

(define (worker-stop [w u32]))

;; ==== private ====

(import start "worker" "start" (i32 i32 i32 i32 i32 i32 i32 i32 i32 i32 -> i32))

(import send "worker" "send" (i32 i32 i32 -> i32))

(import stop "worker" "stop" (i32 ->))
