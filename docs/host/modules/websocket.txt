;; host/modules/websocket
;; WebSocket clients: the host module "websocket". websocket-connect takes options, or 0 for the defaults: the
;; subprotocols to offer and the headers to send, each header's name followed by its value; a page cannot send headers.
;; An open event's bytes are the subprotocol the server chose, empty for none. A message event's value is
;; websocket-text or websocket-binary, which websocket-send also takes.

(const websocket-source u32 7)

(const websocket-open u32 1)

(const websocket-message u32 2)

(const websocket-closed u32 3)

(const websocket-failed u32 4)

(const websocket-drained u32 5)

(const websocket-text i32 1)

(const websocket-binary i32 2)

(struct websocket-options [protocols (slice string)] [headers (slice string)])

(define (websocket-options-init! [o (ptr websocket-options)]))

(define (websocket-connect [url string] [o (ptr websocket-options)] [tag (ptr void)] [err (ptr string)]) -> u32)

(define (websocket-send [id u32] [b (slice u8)] [kind i32]) -> i32)

(define (websocket-send-text [id u32] [s string]) -> i32)

(define (websocket-close [id u32] [code u32]))

;; ==== private ====

(import connect "websocket" "connect" (i32 i32 i32 i32 i32 i32 i32 i32 -> i32))

(import send "websocket" "send" (i32 i32 i32 i32 -> i32))

(import close "websocket" "close" (i32 i32 ->))
