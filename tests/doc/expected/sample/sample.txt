;; sample/sample
;; The sample package's main module: points, and the sizes of shapes.sub's shapes.
;; requires sample/shapes host/fs compiler/macros
;; re-exports (all-from-out shapes)

;; ---- points ----

;; A point on the plane.
(struct point [x f64] [y f64])

(define origin point)

;; distance gives how far apart two points are.
(define (distance [a (ptr point)] [b (ptr point)]) -> f64)

;; ---- sizes ----

(const shape-count i32 3)

;; exported as size
;; area is a shape's width times its height.
(define (area [s (ptr shape)]) -> f64)

;; twice doubles what it is given.
(define-syntax twice)

;; ==== private ====

;; ---- points ----

(define (squared [x f64]) -> f64)

;; ---- sizes ----

;; doubled is twice's transformer.
(define (doubled [stx datum]) -> datum)
