;; sample/generic
;; Generic code: a ring of items, and the largest of some items.
;; requires std/meta/type

;; (ring T N) holds up to N items of type T.
(define-type (ring T [N u32])
  (struct [items (array T N)] [count u32]))

;; (largest T) gives the largest of some items of type T.
(define-generic ((largest T) [xs (slice T)]) -> T)
