;; sample/generic
;; A generic type and a generic function.
;; requires std/meta/generic

;; (ring T N) holds up to N items of type T.
(define-generic (ring T [N u32])
  (struct [items (array T N)] [count u32]))

;; (largest T) finds the largest item in a slice of T.
(define-generic ((largest T) [xs (slice T)]) -> T)
